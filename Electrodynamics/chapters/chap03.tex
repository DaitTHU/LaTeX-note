    多极展开本质是利用逐项逼近的方法来计算电势或能量的，故在系统电荷的空间分布尤为复杂的情况下，使用多极展开去近似计算是一种行之有效的方法。

\end{compactenum}

\section{多极矩展开}
\label{sec:multipole expansion}

考虑分布在有限体积$V$内的电荷系统$\rho(\bm r')$。取以展开原点为心、能包住全部电荷的球，考察球外点$\bm r$的电势。此时对所有源点均有$r>r'$。由$\Phi(\bm r)$的表达式\eqref{eqn:Phi-rho}和$1/\abs{\bm r-\bm r'}$的展开式\eqref{eqn:1/|r-r'|=YY}，代入$r_<=r',\,r_>=r$，有
\begin{align}
    \notag
    \Phi(\bm r)&=\frac1{4\pi\varepsilon_0}\int_V\frac{\rho(\bm r')}{\abs{\bm r'-\bm r}}\d V'\\
    \notag
    &=\frac1{\varepsilon_0}\sum_{\ell=0}^\infty\sum_{m=-\ell}^\ell\frac1{2\ell+1}\biggb{\int_V\rho(\bm r'){r'^\ell}Y\conj_{\ell m}(\theta',\phi')\d V'}\frac{Y_{\ell m}(\theta,\phi)}{r^{\ell+1}}\\
    \label{eqn:Phi multipole}
    &=:\frac1{\varepsilon_0}\sum_{\ell=0}^\infty\sum_{m=-\ell}^\ell\frac{q_{\ell m}}{2\ell+1}\frac{Y_{\ell m}(\theta,\phi)}{r^{\ell+1}}.
\end{align}
其中系数$q_{\ell m}$称为多极矩(multipole moment)
\begin{equation}
    q_{\ell m}:=\int_V\rho(\bm r'){r'^\ell}Y\conj_{\ell m}(\theta',\phi')\d V'.
\end{equation}
另一方面，也可应用式\eqref{eqn:1/|r-r'|}展开$1/\abs{\bm r-\bm r'}$：
\begin{align}
    \notag
    \Phi(\bm r)&=\frac1{4\pi\varepsilon_0}\int_V\rho(\bm r')\sum_{\ell=0}^\infty\frac{r'^\ell}{r^{\ell+1}}P_\ell(\cos\gamma)\d V'\\
    \notag
    &=\frac1{4\pi\varepsilon_0}\biggb{\frac1r\int_V\rho\d V'+\frac1{r^2}\int_V\rho r'\cos\gamma\d V'+\frac1{r^3}\int_V\rho r'^2P_2(\cos\gamma)\d V'+\cdots}\\
    \label{eqn:legendre multipole}
    &=\frac1{4\pi\varepsilon_0}\biggb{\frac qr+\frac{\bm p\cdot\uvec r}{r^2}+\frac1{2r^3}\sum_{\alpha,\beta}Q_{\alpha\beta}\hat r_\alpha\hat r_\beta+\cdots}.
\end{align}
前三个积分项分别对应总电荷即电单极矩(electric monopole moment) $q$、电偶极矩(dipole) $\bm p$、电四极矩(quadrupole) $Q_{\alpha\beta}$：
\begin{subequations}
    \begin{align}
        q={}&\int_V\rho(\bm r')\d V',\\
        \bm p:={}&\int_V\bm r'\rho(\bm r')\d V',\\
        Q_{\alpha\beta}:={}&\int_V(3r_\alpha'r_\beta'-r'^2\delta_{\alpha\beta})\rho(\bm r')\d V'.
    \end{align}
\end{subequations}

% 上式也可通过对$1/\abs{\bm r-\bm r'}$在$\bm r'=0$处Taylor展开推导得到。

\begin{example}{前几个多极矩}{first few multipole moment}
    易得
    \begin{subequations}
        \begin{align}
            q_{00}&=\frac1{\sqrt{4\pi}}\int_V\rho(\bm r')\d V'=\frac1{\sqrt{4\pi}}q;\\
            q_{11}&=-\sqrt{\frac3{8\pi}}\int_V(x'-\i y')\rho(\bm r')\d V'=-\sqrt{\frac3{8\pi}}(p_x-\i p_y),\\
            q_{10}&=\sqrt{\frac3{4\pi}}\int_V z'\rho(\bm r')\d V'=\sqrt{\frac3{4\pi}}p_z;\\
            q_{20}&=\sqrt{\frac5{16\pi}}\int_V(3z'^2-r'^2)\rho(\bm r')\d V'=\sqrt{\frac5{16\pi}}Q_{zz},\\
            q_{21}&=-\sqrt{\frac{15}{8\pi}}\int_V(x'-\i y')z'\rho(\bm r')\d V'=-\sqrt{\frac5{24\pi}}(Q_{xz}-\i Q_{yz}),\\
            q_{22}&=\sqrt{\frac{15}{32\pi}}\int_V(x'-\i y')^2\rho(\bm r')\d V'=\sqrt{\frac5{96\pi}}(Q_{xx}-Q_{yy}-2\i Q_{xy}).
        \end{align}
    \end{subequations}
    负$m$项可由$q_{\ell,-m}=(-)^mq_{\ell m}\conj$推出，
    \tcblower
    对于任何电荷分布，其最低阶的不为0的多极矩$q_{\ell m}$与原点的选择无关，但剩下所有较高阶的多极矩一般都与原点的位置有关。
\end{example}

\begin{example}{等量异号电荷对的电场}{E of pm q}
    考虑位于$z$轴坐标分别为$\pm a$的等量异号电荷对$\pm q$，其在空间激发的电势为
    \begin{align*}
        \Phi&=\frac q{4\pi\varepsilon_0}\biggb{\frac1{\sqrt{a^2+r^2-2ar\cos\theta}}-\frac1{\sqrt{a^2+r^2+2ar\cos\theta}}}\\
        &=\frac q{4\pi\varepsilon_0r}\sum_{\ell=0}^\infty\Bigp{\frac ar}^\ell\bigb{P_{\ell}(\cos\theta)-P_\ell(-\cos\theta)}\\
        &=\frac q{4\pi\varepsilon_0r}\biggb{\frac{2a}r\cos\theta+\Bigp{\frac ar}^3(5\cos^3\theta-3\cos\theta)+\cdots}
    \end{align*}
    对于$r\gg a$的远场，仅保留第一项，与理想电偶极矩$p:=q\cdot 2a$的电势相同：
    \begin{subequations}
        \begin{align}
            \label{eqn:Phi dipole}
            \Phi&=\frac{p\cos\theta}{4\pi\varepsilon_0r^2},\\
            \label{eqn:E dipole}
            \bm E&=\frac p{4\pi\varepsilon_0r^3}(2\cos\theta\uvec r+\sin\theta\uvec\theta).
        \end{align}
    \end{subequations}
    上式只适用于$r\neq0$。若要将理想点偶极子的电场作为全空间上的分布，应由$\bm E=-\grad\Phi$求分布导数。挖去以原点为心的小球并令球半径趋于零，得到球形主值$\PV$和原点处的接触项：
    \begin{equation}
        \label{eqn:E dipole with delta}
        \bm E(\bm r)=\frac1{4\pi\varepsilon_0}\biggb{\PV\frac{3(\bm p\cdot\uvec r)\uvec r-\bm p}{r^3}-\frac{4\pi}3\bm p\delta(\bm r)}.
    \end{equation}
    此时$\div\bm E=-\bm p\cdot\grad\delta(\bm r)/\varepsilon_0$，即电偶极子的电荷密度为
    \begin{equation}
        \rho(\bm r)=-\bm p\cdot\grad\delta(\bm r).
    \end{equation}
\end{example}

\begin{example}
    {电场的体积分}{volume integral of E}
    对于局域分布的电荷系统，其电场在以原点为球心、半径为$R$的球内的体积分为
    \begin{align*}
        \int_{r<R}\bm E(\bm r)\d V&=-\int_{r<R}\grad\Phi(\bm r)\d V=-\int_{r=R}\Phi(\bm r)\d\bm A=-\int_{r=R}\Phi(\bm r)\uvec rR^2\d\Omega\\
        &=-\frac{R^2}{4\pi\varepsilon_0}\int_{r=R}\int_V\frac{\rho(\bm r')}{\abs{\bm r-\bm r'}}\d V'\uvec r\d\Omega,
    \end{align*}
    其中球面上的单位矢量可以写成$Y_{1m}(\theta,\phi)$的线性组合：
    \[
        \uvec r=\sin\theta\cos\phi\uvec x+\sin\theta\sin\phi\uvec y+\cos\theta\uvec z,
    \]
    而$1/\abs{\bm r-\bm r'}$可由式\eqref{eqn:1/|r-r'|=YY}展开，由球谐函数的正交性，级数将只剩$\ell=1$阶项，即
    \[
        \int_{r<R}\bm E(\bm r)\d V=-\frac{R^2}{3\varepsilon_0}\int_V\frac{r_<}{r_>^2}\rho(\bm r')\uvec r'\d V'.
    \]
    其中$r_<:=\min(r',R),\,r_>:=\max(r',R)$。
    \tcblower
    考虑两种极端情形：电荷分布全在球内($r'<R$)和全在球外($r'>R$)，有
    \begin{equation}
        \label{eqn:int E dV}
        \int_{r<R}\bm E(\bm r)\d V=\begin{cases}
            -\frac1{3\varepsilon_0}\bm p,&\text{电荷全在球内};\\[2ex]
            \frac{4\pi}{3}R^3\bm E(\bm 0),&\text{电荷全在球外}.
        \end{cases}
    \end{equation}
    这也对应于电偶极子电场\eqref{eqn:E dipole with delta}中的$-\bm p\delta(\bm r)/3\varepsilon_0$项。
\end{example}

\subsection{能量的多极矩展开}
\label{ssec:multipole expansion of energy}

这里的$\Phi,\bm E$均指外部电荷产生的场，假定在被考察电荷分布所占区域内光滑，且该区域没有外部源。对$\Phi(\bm r)$在原点处Taylor展开
\[
    \Phi(\bm r)=\Phi(\bm 0)-\bm r\cdot\bm E(\bm 0)-\frac12\sum_{\alpha,\beta}x_\alpha x_\beta\pv{E_\beta}{x_\alpha}(\bm 0)+\cdots
\]
由于$\div\bm E=0$，上式可减去$r^2\div\bm E(\bm 0)/6$凑成$Q_{\alpha\beta}$的形式，
由外场相互作用能$W=\int_V\rho\Phi\d V$得到（这里没有自能，故不含$1/2$因子）
\begin{equation}
    W=q\Phi(\bm 0)-\bm p\cdot\bm E(\bm 0)-\frac16\sum_{\alpha,\beta}Q_{\alpha\beta}\pv{E_\beta}{x_\alpha}(\bm 0)+\cdots
\end{equation}
上式体现了多极子和外场的相互作用：电荷与电势、电偶极子与电场、电四极子与电场梯度等。

\begin{example}
    {两个电偶极子之间的互能}{}
    \begin{equation}
        W=\frac{\bm p_1\cdot\bm p_2-3(\uvec n\cdot\bm p_1)(\uvec n\cdot\bm p_2)}{4\pi\varepsilon_0\abs{\bm r_1-\bm r_2}^3},\quad\uvec n:=\frac{\bm r_1-\bm r_2}{\abs{\bm r_1-\bm r_2}}.
    \end{equation}
\end{example}

\paragraph{电偶极子的能量、力矩和力}

对$\bm E$展开至一阶项，得到电偶极子在电场中受到的力为
\begin{equation}
    \label{eqn:F=(p dot nabla)E}
    \bm F=\int_V\rho\bm E\d V\simeq(\bm p\cdot\nabla)\bm E,
\end{equation}
一般地，$\bm p$不随坐标变化，上式也可写成$\grad(\bm p\cdot\bm E)$：
\[
    \grad(\bm p\cdot\bm E)=\bm p\times\cancel{(\curl\bm E)}+\bm E\times\cancel{(\curl\bm p)}+(\bm p\cdot\nabla)\bm E+(\bm E\cdot\cancel{\nabla)\bm p}.
\]
由$\bm F=-\grad U$，可定义电偶极子在电场中的势能为
\begin{equation}
    U=-\bm p\cdot\bm E.
\end{equation}
利用\exmref{exm:E of pm q} 中的模型，可以很直观地推出电偶极子在电场中受到的力矩为
\begin{equation}
    \bm N=\bm p\times\bm E,
\end{equation}

\section{电介质}
\label{sec:dielectric}

电介质(dielectric)是能够被电极化的绝缘体。与导体的自由电荷不同，电介质的带电粒子被原子、分子的内力或分子间的力紧密束缚着，称为束缚电荷(bound charge)。在外电场作用下，这些电荷也只能在微观范围内移动，产生极化(polarization)。
极化主要有三种机制：
\begin{compactenum}
    \item 畸变极化：原子核外的电子云分布产生畸变，从而产生不等于零的电偶极矩；
    \item 位移极化：分子原来正负电中心重合，在外电场作用下正负电中心分离产生电偶极矩；
    \item 转向极化：分子有固有电偶极矩，但取向混乱，宏观上电偶极矩总和等于零。在外电场作用下，各个电偶极子趋向于一致的排列，从而宏观电偶极矩不等于零。
\end{compactenum}
在静电场中，电介质内部可以存在电场，这是电介质与导体的基本区别。
\paragraph{宏观电场}
在前一章中，我们没有考虑有质介质(ponderable media)的存在，所以我们没有区分微观场(microscopic)和宏观场(macroscopic)。
但是静电学主要涉及有质介质中的电荷和场，因此%它们各自的电响应必须考虑在内：(1)分子电荷密度会被扭曲。(2)每个分子的多极矩都是不同的。
我们必须在微观层面上考虑有质介质的极化效应(polarization effect)。为了得到适用于宏观现象的Maxwell方程，我们对宏观小、微观大的区域求平均：% 在第6章中，我们只会小心翼翼地做到这一点
\[
    \bm E:=\ave{\bm E_\text{micro}},
\]
则旋量方程\eqref{eqn:curlE}对求平均后的宏观电场$\bm E$适用：
\begin{equation}
    \label{eqn:curlE-macro}
    \curl\bm E_\text{micro}=\bm 0\implies\curl\bm E=\bm 0.
\end{equation}
这意味着宏观电场$\bm E$同样可以表达成电势的梯度$-\grad\Phi$。

\paragraph{电极化}

%如果电场作用于由大量原子(或分子)组成的介质，束缚在每个分子中的电荷将执行摄动运动(perturbed motions)，由此在介质中产生了电极化。
在电介质内的多极子被外电场极化的过程中，偶极子占主导地位。
\begin{definition}
    {电极化强度}{electric polarization}
    定义电极化强度(electric polarization)为单位体积的电偶极矩
    \begin{equation}
        \bm P:=\sum_i N_i\ave{\bm p_i}.
    \end{equation}
\end{definition}
假设更高阶的宏观多极矩密度可忽略，则有质介质的电势为
\begin{align}
    \notag
    \Phi(\bm r,\bm r')&=\frac1{4\pi\varepsilon_0}\int_V\biggb{\frac{\rho(\bm r')}{\abs{\bm r-\bm r'}}+\frac{\bm P(\bm r')\cdot(\bm r-\bm r')}{\abs{\bm r-\bm r'}^3}}\d V\\
    &=\frac1{4\pi\varepsilon_0}\int_V\frac{\rho(\bm r')-\div\bm P(\bm r')}{\abs{\bm r-\bm r'}}\d V'.
\end{align}
其中$\rho$是自由电荷(或过剩电荷)密度，
$\rho_\pol:=-\div\bm P$是极化电荷密度。%上式是由电荷分布$(\rho-\div\bm P)$激发电势的习惯表达式。
从而
%由$\bm E=-\grad\Phi$，第一个Maxwell方程为
\begin{equation}
    \label{eqn:divE-P}
    \div\bm E=-\laplacian\Phi=\frac{\rho-\div\bm P}{\varepsilon_0}.
\end{equation}
若极化不均匀，则任何小体积内的电荷都会有净增加或净减少，这对应$\div\bm P$这一项。
\begin{definition}
    {电位移矢量}{electric displacement}
    将式\eqref{eqn:divE-P}移项，可定义电位移矢量(electric displacement)
    \begin{equation}
        \bm D:=\varepsilon_0\bm E+\bm P.
    \end{equation}
\end{definition}
从而式\eqref{eqn:divE-P}变为 
\begin{equation}
    \label{eqn:divD}
    \div\bm D=\rho.
\end{equation}
式\eqref{eqn:curlE-macro}和\eqref{eqn:divD}是式\eqref{eqn:curlE}和\eqref{eqn:divE}的宏观对应。

\begin{definition}
    {本构关系}{constitutive relation}
    % 为了解Maxwell方程组，我们需要找到。
    为了简化，我们假设介质是各向同性的(isotropic)，且系统对外加场的响应是线性的(linear)。因此$\bm D$和$\bm E$之间的本构关系(constitutive relation)为
    \begin{equation}
        \bm P=\varepsilon_0\chi_\electric\bm E.
    \end{equation}
    其中无量纲常数$\chi_\electric$称为介质的电极化率(electric susceptibility)。由此得到
    \begin{equation}
        \bm D=\varepsilon\bm E,
    \end{equation}
    其中$\varepsilon:=\varepsilon_\relative\varepsilon_0$称为电容率或介电常数(permittivity)，
    无量纲常数$\varepsilon_\relative:=1+\chi_\electric$称为相对介电常数(relative permittivity)。
\end{definition}

如果电介质不仅是各向同性的，而且是均匀的，则$\varepsilon$与位置无关。式\eqref{eqn:divE-P}可被写为 
\begin{equation}
    \div\bm E=\frac\rho\varepsilon.
\end{equation}
%各向同性
均匀电介质中的所有问题都可归结为前几章的问题，仅需将所有的$\varepsilon_0$替换为$\varepsilon$。

\paragraph{表面分布}

考虑介质界面上$\bm D$和$\bm E$的分布，$\uvec n$是边界上从区域(1)指向区域(2)的法向量，我们可以得到：
\begin{subequations}
    \begin{align}
        \label{eqn:Pn2-Pn1}
        (\bm P_2-\bm P_1)\cdot\uvec n&=-\sigma_\pol,\\
        \label{eqn:Dn2-Dn1}
        (\bm D_2-\bm D_1)\cdot\uvec n&=\sigma,\\
        \label{eqn:Et2-Et1}
        (\bm E_2-\bm E_1)\times\uvec n&=\bm 0.
    \end{align}
\end{subequations}
其中$\sigma_\pol$是边界上表面极化电荷密度；
$\sigma$是宏观表面电荷密度（不包含极化电荷$\sigma_\pol$）。

对于非均匀介质，极化电荷在介质内部随处可见；对于均匀介质，极化电荷仅位于自由电荷附近和两种不同介质之间的界面处；极化电荷所在的界面不是理想的几何界面，实际上，它们的厚度包含了几层分子。

\subsection{含电介质的边值问题}
\label{ssec:boundary-value problem with dielectrics}

我们可以推广之前介绍的方法来解决电介质的边界问题。
\begin{example}{电介质中的镜像电荷}{method of images, dielectrics}
    两种均匀电介质以$z=0$为界，$z>0$一侧的介电常数为$\varepsilon_1$，$z<0$一侧为$\varepsilon_2$，界面没有自由电荷。
    点电荷$q$位于$z$轴上的$z=d$处。场点$(\rho,z)$到真电荷及其镜像位置的距离分别为
    $R_1=\sqrt{\rho^2+(z-d)^2}$和$R_2=\sqrt{\rho^2+(z+d)^2}$。
    \begin{notecaptionblock}
        \includegraphics[width=.98\linewidth]{figures/image_charge_dielectrics.pdf}
        \captionof{figure}[两种电介质界面附近的电场线]{两种电介质中的电场线（$xz$截面）。竖线为界面$z=0$，白点为正电荷$q$，灰点标出计算介质1中的场所用的镜像电荷$q'$的位置。两图取相同的界面交点以比较折角。}
        \label{fig:method of images, dielectrics}
    \end{notecaptionblock}
    为了计算$z>0$部分的电位，在$z<0$部分与$q$对称的位置放置镜像电荷$q'$：
    \[
        \Phi_1=\frac1{4\pi\varepsilon_1}\biggp{\frac q{R_1}+\frac{q'}{R_2}},\quad z>0;
    \]
    为了计算$z<0$部分的电位，在$q$的位置替换为镜像电荷$q''$：
    \[
        \Phi_2=\frac1{4\pi\varepsilon_2}\frac{q''}{R_1},\quad z<0.
    \]
    由边界条件： 
    \begin{align*}
        \begin{cases}
            \varepsilon_1E_{1z}=\varepsilon_2E_{2z}\\
            E_{1\rho}=E_{2\rho}
        \end{cases}\implies
        \begin{cases}
            q-q'=q''\\
            \frac{q+q'}{\varepsilon_1}=\frac{q''}{\varepsilon_2}
        \end{cases}\implies
        \begin{cases}
            q'=-\frac{\varepsilon_2-\varepsilon_1}{\varepsilon_2+\varepsilon_1}q\\[1ex]
            q''=\frac{2\varepsilon_2}{\varepsilon_2+\varepsilon_1}q
        \end{cases}
    \end{align*}
    令$\theta_i$为界面处介质$i$内电场线与法线的夹角，取$0\leq\theta_i<\pi/2$。
    由切向电场和法向电位移连续，有$\tan\theta_2=(\varepsilon_2/\varepsilon_1)\tan\theta_1$。
    因此在\figref{fig:method of images, dielectrics} 中，斜穿界面进入介质2的电场线在左图偏离法线，在右图向法线偏折。在介质2中，电势正比于$1/R_1$，故电场线沿直线延伸。
    \tcblower 
    由式\eqref{eqn:Pn2-Pn1}，$z=0$上极化电荷面密度为
    %\bm P_i=(\varepsilon_i-\varepsilon_0)\bm E_i=-(\varepsilon_i-\varepsilon_0)\grad\Phi(0^\pm).
    \[
        \sigma_\pol=P_{2z}-P_{1z}=-\frac q{2\pi}\frac{\varepsilon_0(\varepsilon_2-\varepsilon_1)}{\varepsilon_1(\varepsilon_2+\varepsilon_1)}\frac d{(\rho^2+d^2)^{3/2}}.
    \]
    当$\varepsilon_2/\varepsilon_1\to\infty$时，$q'\to-q$，介质1内的电场趋于接地导体平面前的电场。
\end{example}
\begin{example}{匀强电场中的介质球}{dielectric sphere in uniform electric field}
    半径为$a$的介质球$\varepsilon$在匀强电场$E_0$中。其电势
    \begin{alignat*}{2}
        \Phi_1&=\sum_{\ell=0}^\infty A_\ell r^\ell P_\ell(\cos\theta),&\quad&r<a\\
        \Phi_2&=\sum_{\ell=0}^\infty\bigb{B_\ell r^\ell+C_\ell r^{-(\ell+1)}}P_\ell(\cos\theta),&&r>a
    \end{alignat*}
    又$r\to\infty$时，$\Phi_2\simeq -E_0r\cos\theta$，故 
    \[
        \Phi_2=-E_0r\cos\theta+\sum_{\ell=0}^\infty C_\ell r^{-(\ell+1)}P_\ell(\cos\theta),
    \]
    由边界条件
    \begin{align*}
        -\biggeval{\frac1r\pv{\Phi_1}\theta}_{r=a}&=-\biggeval{\frac1r\pv{\Phi_2}\theta}_{r=a},\\
        -\varepsilon\biggeval{\pv{\Phi_1}r}_{r=a}&=-\varepsilon_0\biggeval{\pv{\Phi_2}r}_{r=a},
    \end{align*}
    可解得
    \[
        \Phi=\begin{cases}
            -\frac{3}{\varepsilon_\relative+2}E_0r\cos\theta,&r<a\\
            -E_0r\cos\theta+\biggp{\frac{\varepsilon_\relative-1}{\varepsilon_\relative+2}}E_0\frac{a^3}{r^2}\cos\theta,&r>a
        \end{cases}
    \]
    因此介质球内是匀强电场
    \[
        E_1=\frac{3}{\varepsilon_\relative+2}E_0;
    \]
    介质球外相当于外加场$E_0$加上原点的电偶极场，偶极矩朝向电偶极场的方向
    \[
        \bm p=4\pi\varepsilon_0\biggp{\frac{\varepsilon_\relative-1}{\varepsilon_\relative+2}}a^3\bm E_0,
    \]
    球体内的电极化强度
    \begin{equation}
        \label{eqn:P-E}
        \bm P=(\varepsilon-\varepsilon_0)\bm E_1=3\varepsilon_0\biggp{\frac{\varepsilon_\relative-1}{\varepsilon_\relative+2}}\bm E_0,
    \end{equation}
    因此表面极化电荷面密度：
    \[
        \sigma_\pol=\bm P\cdot\uvec r=3\varepsilon_0\biggp{\frac{\varepsilon_\relative-1}{\varepsilon_\relative+2}}E_0\cos\theta.
    \]
\end{example}

\subsection{分子极化率和电极化率}
\label{ssec:molecular polarizability and electric susceptibility}

本节中讨论分子极化率(molecular polarizability) $\gamma_\molecule$和电极化率(electric susceptibility) $\chi_\electric$之间的关系。%我们仅从给它列个大纲开始。

在稀薄(rarefied)介质中，宏观场与作用在任何分子(或分子群)上的场几乎没有区别。在致密(dense)介质中，相邻分子的极化会产生一个内部场$\bm E_\internal$，所以分子所受的局域场为$\bm E_{\mathrm{loc}}=\bm E+\bm E_\internal$。
内部场$\bm E_\internal$除了考虑相邻分子的贡献$\bm E_\near$外，还要减去可以用极化强度$\bm P$描述的平均贡献$\bm E_\pol$：
\[
    \bm E_\internal=\bm E_\near-\bm E_\pol,
\]
由\eqref{eqn:int E dV}，极化强度$\bm P$贡献的平均场为$\bm E_\pol=-\bm P/3\varepsilon_0$；
在立方对称晶格上，球形邻域内等价格点的偶极场相消。各向同性、分子取向关联很弱的介质也可近似取$\bm E_\near=\bm 0$。
故$\bm E_\internal=\bm P/3\varepsilon_0$。

\begin{definition}
    {分子极化率}{molecular polarizability}
    分子极化率描述单个分子在电场下的极化能力
    \begin{equation}
        \ave{\bm p_\molecule}=\varepsilon_0\gamma_\molecule(\bm E+\bm E_\internal).
    \end{equation}
\end{definition}

\begin{theorem}
    {Clausius--Mossotti方程}{Clausius-Mossotti equation}
    分子极化率$\gamma_\molecule$与电极化率$\chi_\electric$之间的关系为
    \begin{equation}
        \gamma_\molecule=\frac3N\frac{\chi_\electric}{\chi_\electric+3}=\frac3N\frac{\varepsilon_\relative-1}{\varepsilon_\relative+2}.
    \end{equation}
\end{theorem}

\begin{proof}
    由电极化强度的定义
    \[
        \bm P=N\ave{\bm p_\molecule}=N\varepsilon_0\gamma_\molecule(\bm E+\bm E_\internal)=N\gamma_\molecule\biggp{\varepsilon_0\bm E+\frac{\bm P}3}.
    \]
    代入$\bm P=\varepsilon_0\chi_\electric\bm E,\enspace\varepsilon_\relative=1+\chi_\electric$即得。
\end{proof}

\begin{remark}
    这种关系对气体等稀释物质最适用。
    对于液体和固体，尤其是$\varepsilon_\relative$很大时，上式仅近似有效。
\end{remark}

\paragraph{分子极化率的模型}
把分子内的束缚电荷看作固有频率为$\omega_0$的谐振子。外场使其平衡位置偏移，产生感应偶极矩
\[
    m\omega_0^2\bm r=e\bm E_{\mathrm{loc}},\qquad
    \bm p_{\mathrm{ind}}=e\bm r=\frac{e^2}{m\omega_0^2}\bm E_{\mathrm{loc}}.
\]
若分子还有大小为$p_0$的永久偶极矩，外场也会改变分子的取向。设分子能自由转动，彼此的取向关联可忽略。偶极矩与局域场成$\theta$角时，势能为$-p_0E_{\mathrm{loc}}\cos\theta$。在温度$T$下，记$x=p_0E_{\mathrm{loc}}/(\kB T)$，对取向的Boltzmann权重积分得
\[
    Z(x)=\int\e^{x\cos\theta}\d\Omega
    =2\pi\int_{-1}^1\e^{xu}\d u
    =4\pi\frac{\sinh x}{x}.
\]
垂直于局域场的分量平均为零，沿场方向的平均分量为
\begin{equation}
    \ave{p_\parallel}=p_0\dv{\ln Z}{x}
    =p_0\biggp{\coth x-\frac1x}.
\end{equation}
括号中的函数称为Langevin函数。弱场下$x\ll1$，它的首项是$x/3$，因而$\ave{p_\parallel}\simeq p_0^2 E_{\mathrm{loc}}/(3\kB T)$。将这一项与感应偶极矩相加，按本节的定义得到静态分子极化率
\begin{equation}
    \gamma_\molecule\simeq\frac{e^2}{m\omega_0^2\varepsilon_0}+\frac{p_0^2}{3\varepsilon_0\kB T}.
\end{equation}
在固定分子数密度$N$的稀薄气体中，$\chi_\electric\simeq N\gamma_\molecule$。降温使取向极化增强。固定温度而增大场强时，$\ave{p_\parallel}$趋于$p_0$，永久偶极对极化强度的贡献趋于$Np_0$。

\begin{example}
    {}{}
    常温常压(normal temperature and pressure, NTP)下，许多气体的$\chi_\electric<10^{-3}$，而常见固体和液体的$\chi_\electric$可达$1$量级或更大。
\end{example}

\subsection{电介质中的静电能}
\label{ssec:electrostatic energy in dielectric media}

固定电极和介质，准静态地把自由电荷从零增加到终值。将电荷$\delta q$送到电势$\Phi$处，电源做功$\Phi\,\delta q$。用Gauss定律，并对全空间分部积分，可得
\[
    \delta W=\int\Phi\,\delta\rho_{\mathrm f}\d V
        =\int\bm E\cdot\delta\bm D\d V .
\]
这里假设电荷局域分布，电势在无穷远趋于零，导体的面电荷也计入$\rho_{\mathrm f}$。对于固定材料状态的可逆响应，以无场态为能量零点，静电能密度为
\[
    w_\electric=\int_{\bm 0}^{\bm D}\bm E(\bm D')\cdot\d\bm D'.
\]
若$\bm D=\varepsilon\bm E$，或更一般地介电张量线性且对称，则
\begin{equation}
    W=\frac12\int\bm E\cdot\bm D\d V=\frac12\int\rho_{\mathrm f}\Phi\d V .
\end{equation}

\paragraph{给定自由电荷分布时的能量变化}
在原来的真空场$\bm E_0$中放入线性电介质，保持自由电荷的空间分布不变。此时$\div(\bm D-\bm D_0)=0$。
分别取插入前后的电势作试探函数$\psi=\Phi_0,\Phi$，分部积分给出
\[
    \int(-\grad\psi)\cdot(\bm D-\bm D_0)\d V
        =\int\psi\,\div(\bm D-\bm D_0)\d V=0 .
\]
这样可以在全空间积分中将$\bm E\cdot\bm D$换成$\bm E\cdot\bm D_0$，将$\bm E_0\cdot\bm D_0$换成$\bm E_0\cdot\bm D$。用$\bm D_0=\varepsilon_0\bm E_0$和$\bm D=\varepsilon_0\bm E+\bm P$得
\begin{equation}
    \Delta W=\frac12\int(\bm E\cdot\bm D_0-\bm E_0\cdot\bm D)\d V
            =-\frac12\int_{\text{介质}}\bm P\cdot\bm E_0\d V .
\end{equation}
这个积分给出两种构型的总储能之差。要问插入后某处有多少储能，仍应使用该处的$\bm E\cdot\bm D/2$。

\begin{example}{电容器吸入介质片}{dielectric insertion work}
    平行板长$\ell$、宽$b$、间距$d$。介电常数为$\varepsilon>\varepsilon_0$的片材填满间隙，沿板长插入距离$x$。忽略边缘场对电容的修正，已填充和未填充部分并联，故
    \[
        C(x)=\frac b d\bigb{\varepsilon x+\varepsilon_0(\ell-x)} .
    \]
    断开电源后，每块极板的总电荷固定，面电荷可以重新分布。场对介质做的功来自储能的减少，沿插入方向的力为
    \[
        F_Q=-\pp x\biggp{\frac{Q^2}{2C(x)}}_Q
            =\frac{Q^2}{2C(x)^2}\dv Cx .
    \]
    接回电源并保持电压$V$时，极板又得到电荷$\d Q=V\d C$。电源做功$V\d Q$，其中一半成为新增的场能，另一半成为机械功，
    \[
        F_V\d x=V\d Q-\d\biggp{\frac12 CV^2}=\frac12 V^2\d C,\qquad
        F_V=\frac{b(\varepsilon-\varepsilon_0)}{2d}V^2 .
    \]
    在同一位置、同一瞬时电压下代入$Q=CV$，两种算法都给出指向电容器内部的力。
\end{example}
