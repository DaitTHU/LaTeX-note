\section{Foundamental Posulates}
\subsection{State Desription}
Quantum state\index{quantum state} is a vector $\ket\psi$ in Hilbert space.
\begin{definition}{Dirac Notation}{}
	The \textbf{ket} $\ket\psi$ is the $n\times 1$ column vector, and the \textbf{bra} $\bra\psi=\ket\psi\adjoint$,
	$$\ket\psi=[\psi_1~\cdots~\psi_n]\transp,\qquad
		\bra\psi=[\psi_1\conj~\cdots~\psi_n\conj].$$
	The inner product\index{inner product} of two vectors $\ket a$ and $\ket b$ is
	\begin{center}
		$\brkt ab:=a_1\conj b_1+\cdots+a_n\conj b_n=\sum_{i=1}^na_i\conj b_i$.\footnote[1]{I'll omit the upper and lower mark for simplicity.}
	\end{center}
	The quantum state should be \textbf{normalized}, i.e. $\brkt\psi\psi=1$.
\end{definition}
%Hilbert space is the complete inner product space, and 
\begin{theorem}{Gram-Schmidt}{}
	Given a linearly independent bases $\ket{v_1},\ldots,\ket{v_n}$, we can form linear combinations of the basis vectors to obtain an orthonormal basis.\\
	Thus we could find a set of orthonormal bases $\ket{a_i}$, and
	$$\brkt{a_i}{a_j}=\delta_{ij}\qquad\sum\ktbr{a_i}{a_i}=I.$$
	In the $\ket{a_i}$ base, the representation of a vector is $\ket\psi=\sum\psi_i\ket{a_i}$.
\end{theorem}
$\brkt x\psi=\psi(x)$ is the wave function.\index{wave function}

\subsection{Measurable Physical Properties}
Measurable physical properties can be represented by Hermite\index{Hermite} operator $\op{A}$.
\begin{theorem}{Eigenvalues of Hermitian}{}
	The eigenvalues\index{eigenvalue} of Hermite $\op{A}$ are \textbf{real}, because if $\op{A}\ket a=a\ket a$,
	\begin{align*}
		\bra a\op{A}\ket a\adjoint & =\bra a\op{A}\ket a,\footnotemark \\
		a\conj\brkt aa             & =a\brkt aa,
	\end{align*}\footnotetext[1]{Here the dagger symbol acts on the whole bracket $\bra a\op{A}\ket a$.}
	thus $a\in\RR$.
\end{theorem}
\begin{theorem}{Eigenvectors of Hermitian}{}
	The eigenvectors\index{eigenvector} corresponding to different eigenvalues are \textbf{orthogonal}, because if $\op{A}\ket{a_1}=a_1\ket{a_1},\op{A}\ket{a_2}=a_2\ket{a_2}$
	\begin{align*}
		\bra{a_2}\op{A}\ket{a_1}\adjoint & =\bra{a_1}\op{A}\ket{a_2} \\
		a_1\conj\brkt{a_1}{a_2}          & =a_2\brkt{a_1}{a_2},
	\end{align*}
	for $a_1\conj=a_1\neq a_2$, $\brkt{a_1}{a_2}=0$.
\end{theorem}
\subsubsection{Position Operator \textsf{X}}
Position operator $\op{X}$\index{position operator $\op{X}$} in $\ket x$ base\index{base} satisfies: every position $\ket x$ is an eigenvector with its position $x$ as the eigenvalue, thus
$$\op{X}\ket x=x\ket x.$$
The base $\ket x$ is continuous, where $x\in\RR$, and is orthonormal
$$\brkt{x'}x=\delta(x'-x),\qquad\intR \ko2\ktbr xx\d x=I.$$
$\op{X}\ket\psi$ is a new state, which could be represented as
$$\bra x\op{X}\ket\psi=x\brkt x\psi=x\psi(x).$$
\begin{example}{Verifying the Hermitian}{}
	$\op{X}$ is Hermite because
	\begin{align*}
		\bra\varphi\op{X}\ket\psi & =\int\brkt\varphi{x}\ko3\bra x\op{X}\ket\psi\d x                               \\
		                       & =\int\brkt x\varphi x\brkt x\psi\d x=\int x\varphi\conj(x)\psi(x)\d x;         \\
		\bra\psi\op{X}\ket\varphi & =\int x\psi\conj(x)\varphi(x)\d x=\bra\varphi\op{X}\ket\psi\adjoint.\quad(x\conj=x)
	\end{align*}
\end{example}
\subsubsection{Momentum Operator \textsf{P}}
Momentum operator $\op{P}$\index{momentum operator $\op{P}$} in $\ket p$ base, similarily
$$\op{P}\ket p=p\ket p.$$
%To derive the expression in $\ket x$ base,
We consider the state $\ket\psi$ in $\ket x$, $\ket k$ base, that $\brkt x\psi=\psi(x)$, $\brkt k\psi=:\hat\psi(k)$. Then by the \textbf{Fourier Transfromation}\index{Fourier Transfromation}:
$$\psi(x)=\FT\inv[\hat\psi(k)]=\frac1{\sqrt{2\pi}}\int\hat\psi(k)\exp(\i kx)\d k.$$
According to the \textbf{de Broglie relation}\index{de Broglie relation}: $p=\hbar k$, thus $\brkt k\psi\propto\brkt p\psi=:\varphi(p)$,
\begin{align*}
	\psi(x)     & \propto\int\varphi(p)\exp\Bigp{\i\frac p\hbar x}\d p, \\
	\brkt x\psi & =\int\brkt xp\ko3\brkt p\psi\d p,
\end{align*}
the lower formula is the calculation of $\brkt x\psi$, comparing these two formulas, we could conclude that the eigenfunction\index{eigenfunction} $p(x)$ is
$$p(x)=\brkt xp\propto\exp\Bigp{\frac{\i p}\hbar x},$$
which meets the equation
$$\dv{\brkt xp}x=\frac{\i p}\hbar\brkt xp,$$
From the definition $\op{P}\ket p=p\ket p$, we have the eigenfunction $p(x)$ satisfies
$$\bra x\op{P}\ket p=p\brkt xp=-\i\hbar\dv{\brkt xp}x.$$
Therefore, the momentum operator $\op{P}$ in $\ket x$ base is $\op{P}\to-\i\hbar\d/\nd x,$
$$\bra x\op{P}\ket\psi=-\i\hbar\dd x\brkt x\psi=-\i\hbar\dv{\psi(x)}x.$$
\begin{example}{Verifying the Hermitian}{}
	$\op{P}$ is Hermite because
	\begin{align*}
		\bra\varphi\op{P}\ket\psi & =\int\brkt\varphi{x}\ko3\bra x\op{P}\ket\psi\d x=-\i\hbar\int\varphi\conj(x)\dv{\psi(x)}x\d x                       \\
		                       & =-\i\hbar\biggl[\cancel{\varphi\conj(x)\psi(x)\Big|_{-\infty}^{+\infty}}-\int\psi(x)\dv{\varphi\conj(x)}x\d x\biggr] \\
		                       & =\i\hbar\int\psi(x)\dv{\varphi\conj(x)}x\d x=\bra\psi\op{P}\ket\varphi\adjoint.
	\end{align*}
	Note: $x\to\infty,\quad\psi(x),\varphi(x)\to0$
\end{example}
\begin{definition}{Commutator}{}
	The commutator\index{commutator} of two operators is
	$$[\op{A},\op{B}]:=\op{A}\op{B}-\op{B}\op{A}.$$
	And anti-commutator\index{anti-commutator} $\{\op{A},\op{B}\}:=\op{A}\op{B}+\op{B}\op{A}$, which is also useful later.
\end{definition}
\begin{example}{$[\op{X},\op{P}]$}{}\su{12}
	\begin{align*}
		\bra x[\op{X},\op{P}] & \ket\psi=\bra x\op{X}\op{P}\ket\psi-\bra x\op{P}\op{X}\ket\psi
		=x\bra x\op{P}\ket\psi-\bra x\op{P}\,\big(\,\op{X}\ket\psi\big)                                                                     \\
		=                   & -\i\hbar\Bigl(x\dv{\brkt x\psi}x-\dv{\bra x\op{X}\ket\psi}x\Bigr)
		=-\i\hbar\Bigl(x\dv{\brkt x\psi}x-\dv{x\brkt x\psi}x\Bigr)                                                                 \\
		=                   & -\i\hbar\biggl[\cancel{x\dv{\brkt x\psi}x}-\Bigl(\cancel{x\dv{\brkt x\psi}x}+\brkt x\psi\Bigr)\biggr]
		=\i\hbar\brkt x\psi,
	\end{align*}
	then we conclude that
	$$\spark{[\op{X},\op{P}]=\i\hbar.}$$
\end{example}

\subsubsection{Angular Momentum Operator \textsf{L}}
The classical angular momentum is
$$\bm L=\bm r\times\bm p.$$
In the quantum, the momentum $\op{P}=-\i\hbar\nabla$, and\index{angular momentum operator $\op{L}$}
\begin{gather*}
	\op{L}=-\i\hbar\,\bm r\times\nabla\\
	=-\i\hbar\Bigb{\,y\pp z\ko2-\ko2z\pp y\,,\,z\pp x\ko2-\ko2x\pp z\,,\,x\pp y\ko2-\ko2y\pp x\,}\transp,
\end{gather*}
$\op{L}_x,\op{L}_y,\op{L}_z$ are three portions of $\op{L}$, and the angular momentum squared is
$$\op{L}^2=\op{L}_x^2+\op{L}_y^2+\op{L}_z^2.$$
\iffalse

	\begin{align*}

		\op{L}^2   & =-\hbar^2\Bigl(y^2\pp[2]z+z^2\pp[2]y+z^2\pp[2]x+x^2\pp[2]z+z^2\pp[2]y+y^2\pp[2]z \\
		        & \qquad\qquad-y\pp y-z\pp z-z\pp z-x\pp x-x\pp x-y\pp y                            \\
		        & \qquad\qquad-2yz\pw{}yz-2zx\pw{}zx-2xy\pw{}xy\Bigr)
	\end{align*}

\begin{example}{$[\op{L}_x,\op{L}_y]~\&~[\op{L}^2,\op{L}_x]$}{}\su{10}
	\begin{gather*}
		\begin{aligned}
			\op{L}_x\op{L}_y= & -\hbar^2\Bigp{y\pp x+\cancel{yz\pw{}zx}-\cancel{xy\pp[2]z}-\cancel{z^2\pw{}yx}+\cancel{xz\pw{}yz}}; \\
			\op{L}_y\op{L}_x= & -\hbar^2\Bigp{\cancel{yz\pw{}xz}-\cancel{z^2\pw{}xy}-\cancel{xy\pp[2]z}+x\pp y+\cancel{xz\pw{}zy}}, %\footnotemark
		\end{aligned}\\
		[\op{L}_x,\op{L}_y]=-\hbar^2\Bigp{y\pp x-x\pp y}=\i\hbar\,\op{L}_z.
	\end{gather*}
	%\footnotetext[1]{$$\partial^2/\partial x\partial y\equiv\pw{}yx$$}
	Similarily,
	$$\spark{[\op{L}_x,\op{L}_y]=\i\hbar\,\op{L}_z\quad[\op{L}_y,\op{L}_z]=\i\hbar\,\op{L}_x,\quad[\op{L}_z,\op{L}_x]=\i\hbar\,\op{L}_y.}$$
	Then calculate $[\op{L}^2,\op{L}_x]=[\op{L}_x^2,\op{L}_x]+[\op{L}^2_y,\op{L}_x]+[\op{L}^2_z,\op{L}_x]$,
	\begin{align*}
		[\op{L}_x^2,\op{L}_x] & =\op{L}_x^3-\op{L}_x^3=0,                                                     \\
		[\op{L}^2_y,\op{L}_x] & =\op{L}_y[\op{L}_y,\op{L}_x]+[\op{L}_y,\op{L}_x]\op{L}_y                          \\
		                    & =-\i\hbar\,\op{L}_y\op{L}_z-\i\hbar\,\op{L}_z\op{L}_y=-\i\hbar\,\{\op{L}_y,\op{L}_z\},       \\
		[\op{L}^2_z,\op{L}_x] & =\op{L}_z[\op{L}_z,\op{L}_x]+[\op{L}_z,\op{L}_x]\op{L}_z=\i\hbar\,\{\op{L}_z,\op{L}_y\}.
	\end{align*}
	Thus
	$$[\op{L}^2,\op{L}_x]=0-\i\hbar\,\{\op{L}_y,\op{L}_z\}+\i\hbar\,\{\op{L}_z,\op{L}_y\}=0.$$
	Similarily,
	$$\spark{[\op{L}^2,\op{L}_x]=[\op{L}^2,\op{L}_y]=[\op{L}^2,\op{L}_z]=0.}$$
\end{example}
\begin{definition}{Ladder Operator}{}
	Define the useful ladder operator\index{ladder operator $\op{L}_\pm$} $\op{L}_\pm:=\op{L}_x\pm \i\,\op{L}_y.$
	\begin{example}{Denote $\op{L}^2$ by $\op{L}_\pm~\&~\op{L}_z$}{}
		$\op{L}^2=\op{L}_\pm\op{L}_\mp+\op{L}_z^2\mp\hbar\,\op{L}_z$, because
		\begin{align*}
			\op{L}_\pm\op{L}_\mp & =(\op{L}_x\pm \i\,\op{L}_y)(\op{L}_x\mp \i\,\op{L}_y)                                      \\
			               & =\op{L}_x^2+\op{L}_y^2\mp \i\,(\op{L}_x\op{L}_y-\op{L}_y\op{L}_x)=\op{L}^2-\op{L}_z^2\pm\hbar\,\op{L}_z.
		\end{align*}
	\end{example}
	From the definition, its commutator with $\op{L}_z,\op{L}^2$ is
	\iffalse
		\begin{align*}
			[\op{L}_z,\op{L}_\pm] & =[\op{L}_z,\op{L}_x]\pm \i[\op{L}_z,\op{L}_y]                              \\
			                & =\i\hbar\op{L}_y\pm \i(-\i\hbar\op{L}_x)=\pm\hbar(\op{L}_x\pm \i\op{L}_y).
		\end{align*}
		Thus\fi
	$$\spark{[\op{L}_z,\op{L}_\pm]=\pm\hbar\,\op{L}_\pm,\quad[\op{L}^2,\op{L}_\pm]=0.}$$
	As $[\op{L}^2,\op{L}_z]=0$, $\op{L}_2,\op{L}_z$ have the same eigenfunction $\ket\psi$, i.e.
	$$\op{L}^2\ket\psi=\lambda\ket\psi,\quad\op{L}_z\ket\psi=\mu\ket\psi.$$
	Considering $\op{L}_\pm\ket\psi$,
	$$\op{L}^2\op{L}_\pm\ket\psi=\op{L}_\pm\op{L}^2\ket\psi=\lambda\op{L}_\pm\ket\psi,$$
	$\ket\psi$, $\op{L}_\pm\ket\psi$ share the \textbf{same} eigenvalue of $\op{L}^2$.
	$$\op{L}_z\op{L}_\pm\ket\psi=(\op{L}_\pm\op{L}_z\pm\hbar\,\op{L}_\pm)\ket\psi=(\mu\pm\hbar)\op{L}_\pm\ket\psi.$$
	We call $\op{L}_+$ the \textbf{rasing operator}, as it increases the eigenvalue of $\op{L}_z$ by $\hbar$, and $\op{L}_-$ the \textbf{lowering operator}.
\end{definition}
\noindent Simplify $\ket{\psi_n}:=\op{L}_+^n\ket\psi$, ($n\geqslant 0$)
$$\ave{\op{L}^2}=\bra{\psi_n}\op{L}^2\ket{\psi_n}=\lambda.$$
$$\ave{\op{L}_z^2}=\bra{\psi_n}\op{L}_z^2\ket{\psi_n}=(\mu+n\hbar)^2.$$
While
$$\ave{\op{L}^2}=\ave{\op{L}_x^2}+\ave{\op{L}_y^2}+\ave{\op{L}_z^2}\geqslant\ave{\op{L}_z^2}.$$
Hence, the rising progress can't go on forever, there must exist a \textbf{top} $\ket{\psi_t}$:
$$\op{L}_+\ket{\psi_t}=0,\qquad{\rm then}~\ket{\psi_n}\equiv 0,~\forall\,n>t.$$
Let $\ell\hbar$ be the eigenvalue of $\op{L}_z$ at $\ket{\psi_t}$, i.e. $\op{L}_z\ket{\psi_t}=\ell\,\hbar\ket{\psi_t}$.
\begin{align*}
	\op{L}^2\ket{\psi_t} & =(\op{L}_-\op{L}_++\op{L}_z^2+\hbar\,\op{L}_z)\ket{\psi_t}                                \\
	                  & =(0+\ell^2\hbar^2+\ell\,\hbar^2)\ket{\psi_t}=\ell(\ell+1)\hbar^2\ket{\psi_t}.
\end{align*}
Also, there exists a \textbf{bottom} $\ket{\psi_b}$ that $\op{L}_-\ket{\psi_b}=0$, $\op{L}_z\ket{\psi_b}=\jmath\,\hbar\ket{\psi_b}$
\begin{align*}
	\op{L}^2\ket{\psi_b} & =(\op{L}_+\op{L}_-+\op{L}_z^2-\hbar\op{L}_z)\ket{\psi_b}                                          \\
	                  & =(0+\jmath^2\hbar^2-\jmath\,\hbar^2)\ket{\psi_b}=\jmath(\jmath-1)\hbar^2\ket{\psi_b}.
\end{align*}
Because $\forall\,n,\op{L}^2\ket{\psi_n}\equiv\lambda\ket{\psi_n}$,
$$\lambda=\ell(\ell+1)\hbar^2=\jmath(\jmath-1)\hbar^2\quad\Rightarrow\quad\jmath=-\ell\text{~or~}\bcancel{\jmath=\ell+1}.$$
%(bottom$\,>\,$top, absurd).
While $\op{L}_z\ket\psi=m\hbar\ket\psi$, where $m=-\ell,\ldots,\ell$ in $N$ integer steps, hence, $2\ell\in\NN$. $\ket\psi$ contains two numbers $\ell,m$, using the notation $\ket{\ell,m}:=\ket\psi$ is more clear for different $\ket\psi$,
$$\spark{\op{L}^2\ket{\ell,m}=\ell(\ell+1)\hbar^2\ket{\ell,m},\quad\op{L}_z\ket{\ell,m}=m\hbar\ket{\ell,m}.}$$
where $\ell=0,1/2,1,3/2,\ldots;m=-\ell,-\ell+1,\ldots,\ell$.
\begin{example}{$\op{L}_\pm$ changes $m$}{}
	$\op{L}_\pm$ changes the value of $m$, i.e.
	$$\op{L}_+\ket{\ell,m}=\alpha\ket{\ell,m+1},\quad\op{L}_-\ket{\ell,m+1}=\beta\ket{\ell,m},$$
	We set $\alpha,\beta\in\RR_+$.
	$$(\op{L}_+\ket{\ell,m})\adjoint=\bra{\ell,m}\op{L}_-=\alpha\bra{\ell,m+1},$$
	right multiply $\ket{\ell,m+1}$,
	\begin{align*}
		\bra{\ell,m}\op{L}_-\ket{\ell,m+1}=\beta & \brkt{\ell,m}{\ell,m}=\alpha\brkt{\ell,m+1}{\ell,m+1}, \\
		\Rightarrow                           & \quad\alpha=\beta.
	\end{align*}
	%Right multiply $\op{L}_\pm\op{L}_\mp=\op{L}^2-\op{L}_z^2\pm\hbar\,\op{L}_z$ by $\ket{\ell,m}$,
	\begin{align*}
		\op{L}_-\op{L}_+\ket{\ell,m} & =(\op{L}^2-\op{L}_z^2-\hbar\,\op{L}_z)\ket{\ell,m}   \\
		\alpha^2\ket{\ell,m}   & =[\ell(\ell+1)-m(m+1)]\hbar^2\ket{\ell,m}.
	\end{align*}
	$$\Rightarrow\quad\spark{\op{L}_\pm\ket{\ell,m}=\sqrt{\ell(\ell+1)-m(m\pm 1)}\hbar\ket{\ell,m\pm 1}}.$$
\end{example}
\paragraph{Spherical Expression}The nabla in spherical coordinate,\index{spherical coordinate}
$$\nabla=\pp r\uvec r+\frac1r\pp \theta\uvec\theta+\frac1{r\sin\theta}\pp \phi\uvec\phi.$$
Because $\bm r=r\uvec r$, and $\uvec{r}\times\uvec{r}=\bm 0,\uvec{r}\times\uvec{\theta}=\uvec{\phi},\uvec{r}\times\uvec{\phi}=-{\uvec{\theta}}$,
$$\op{L}=-\i\hbar\cdot r\uvec r\times\nabla=-\i\hbar\Bigp{\pp \theta\uvec\phi-\frac1{\sin\theta}\pp \phi\uvec\theta},$$
Back to Cartesian components,
\begin{align*}
	\uvec\theta & =\cos\theta\cos\phi\uvec i+\cos\theta\sin\phi\uvec j-\sin\theta\uvec k, \\
	\uvec\phi   & =-\sin\phi\uvec i+cos\phi\uvec j.
\end{align*}
Evidently,
\begin{align*}
	\begin{aligned}
		\op{L}_x & =+\i\hbar\Bigp{\sin\phi\pp \theta+\cos\phi\cot\theta\pp \phi}, \\
		\op{L}_y & =-\i\hbar\Bigp{\cos\phi\pp \theta-\sin\phi\cot\theta\pp \phi},
	\end{aligned}\quad
	\spark{\op{L}_z=-\i\hbar\pp \phi.}
\end{align*}
Then,
\begin{gather*}
	\op{L}_\pm=\op{L}_x\pm \i\,\op{L}_y=\hbar\e^{\pm \i\phi}\Bigp{\pm\pp \theta+\i\cot\phi\pp \phi}.\\
	\op{L}_+\op{L}_-=-\hbar^2\Bigp{\pp[2]\theta+\cot\theta\pp \theta+\cot^2\theta\pp[2]\phi+\i\pp \phi}.\\
	\spark{\op{L}^2=\op{L}_+\op{L}_-+\op{L}_z^2-\hbar\,\op{L}_z=-\hbar^2\Lambda^2,}
\end{gather*}
where the Legendrian\index{Legendrian $\Lambda^2$}
$$\Lambda^2:=\frac1{\sin\theta}\pp \theta\Bigl(\sin\theta\pp \theta\Bigr)+\frac1{\sin^2\theta}\pp[2]\phi,$$
the eigenfunction of $\op{L}^2$, i.e.
$$\op{L}^2\psi=-\hbar^2\Lambda^2\psi=\lambda\psi,$$
\setcounter{footnote}{0}
is the Legendre function we'll solve in H-Atom.\footnote{Note, parenthetically, that eigenfunctions of $\op{L}^2$ have been known since the 19th century, long before quantum mechanics was born.}\clearpage
\subsubsection{Function of Operator}
Using the \textbf{Taylor Expansion}\index{Taylor Expansion}
$$f(x)=\sum_{k=0}^\infty\frac{f^{(k)}(0)}{k!}x^k=f(0)+f'(0)x+\frac12f''(0)x^2+\cdots,$$
just replace $x$ by $\op{A}$,
$$f(\op{A})=\sum_{k=0}^\infty\frac{f^{(k)}(0)}{k!}\op{A}^k=f(0)+f'(0)\op{A}+\frac12f''(0)\op{A}^2+\cdots.$$
\begin{theorem}{About eigenvectors}{}
	If $\op{A}\ket{a_i}=a_i\ket{a_i}$, then \su3
	$$f(\op{A})\ket{a_i}=f(a_i)\ket{a_i}.$$
	Because $\op{A}^n\ket{a_i}=a_i^n\ket{a_i}$ and $f(\op{A})=\sum c_n\op{A}^n$
	$$f(\op{A})\ket{a_i}=\sum c_n\op{A}^n\ket{a_i}=f(a_i)\ket{a_i}.$$
	thus $f(\op{A})=\sum f(\op{A})\ktbr{a_i}{a_i}=\sum f(a_i)\ktbr{a_i}{a_i}$.
\end{theorem}
\begin{example}{$\op{K}~\&~V(\op{X})$}{}
	Kinetic energy\index{kinetic energy $\op{K}$} $\op{K}:=K(\op{P})=\frac{\op{P}^2}{2m}$, and potential energy function\index{potential $V$} $V(\op{X})$\su5
	\begin{align*}
		\bra x\op{K}\ket\psi
		=                     & \frac1{2m}\bra x\op{P}^2\ket\psi
		=\frac1{2m}\bra x\op{P}(\op{P}\ket\psi)                                                                   \\
		=                     & \frac{\hbar}{2\i m}\dv{\bra x\op{P}\ket\psi}x=-\frac{\hbar^2}{2m}\dv[2]{\psi(x)}x. \\
		\bra xV(\op{X})\ket\psi= & \int\bra xV(\op{X})\ket{x'}\brkt{x'}\psi\d{x'}                                 \\
		=                     & \int V(x')\brkt x{x'}\psi(x')\d{x'}=V(x)\psi(x).
	\end{align*}
	And the Hamiltonian\index{Hamiltonian $\op{H}$} $\op{H}=\op{K}+V(\op{X})$.
\end{example}
\begin{theorem}{About Commutator}{}
	Commutator is anti-Hermite, because
	$$[\op{A},\op{B}]\adjoint= \op{B}\adjoint\op{A}\adjoint-\op{A}\adjoint\op{B}\adjoint=\op{B}\op{A}-\op{A}\op{B}=-[\op{A},\op{B}],$$
	The first thing about commutator is that
	$$[\op{A},\op{A}^n]=0,\qquad\forall\,n\in\NN.$$
	Therefore, $$[\op{A},f(\op{A})]=0.$$\tcblower
	Commutator is much like cross product for they both satisfy the \textbf{inverse exchange law}:\index{inverse exchange law}
	$$[\op{B},\op{A}]=-[\op{A},\op{B}]\quad\leftrightarrow\quad\bm b\times\bm a=-\bm a\times\bm b.$$
	In cross product, we have the Lagrange equation:
	\begin{align*}
		(\bm a\times\bm b)\times\bm c & =(\bm a\cdot\bm c)\,\bm b-\bm a\,(\bm b\cdot\bm c); \\
		\bm a\times(\bm b\times\bm c) & =\bm b\,(\bm a\cdot\bm c)-(\bm a\cdot\bm b)\,\bm c.
	\end{align*}
	In the commutator, the relation is similar:
	\begin{itemize}
		\item $[\op{A}\op{B},\op{C}]=[\op{A},\op{C}]\op{B}+\op{A}[\op{B},\op{C}]$
		      \begin{align*}
			      [\op{A}\op{B},\op{C}] & =\op{A}\op{B}\op{C}-\op{C}\op{A}\op{B}                                                        \\
			                       & =\op{A}\op{B}\op{C}-\op{A}\op{C}\op{B}+\op{A}\op{C}\op{B}-\op{C}\op{A}\op{B}=[\op{A},\op{C}]\op{B}+\op{A}[\op{B},\op{C}].
		      \end{align*}
		\item $[\op{A},\op{B}\op{C}]=\op{B}[\op{A},\op{C}]+[\op{A},\op{B}]\op{C}$
		      \begin{align*}
			      [\op{A},\op{B}\op{C}] & =\op{A}\op{B}\op{C}-\op{B}\op{C}\op{A}                                                        \\
			                       & =\op{B}\op{A}\op{C}-\op{B}\op{C}\op{A}+\op{A}\op{B}\op{C}-\op{B}\op{A}\op{C}=\op{B}[\op{A},\op{C}]+[\op{A},\op{B}]\op{C}.
		      \end{align*}
	\end{itemize}
	Let $\op{B}=\op{C}$, then
	$$[\op{A},\op{B}^2]=\op{B}[\op{A},\op{B}]+[\op{A},\op{B}]\op{B}=\bigl\{[\op{A},\op{B}],\op{B}\bigr\}.$$
	Especially, when $\op{B}[\op{A},\op{B}]=[\op{A},\op{B}]\op{B}$,
	\begin{align*}
		\bullet~[\op{A},\op{B}^n]= & [\op{A},\op{B}^{n-1}]\op{B}+[\op{A},\op{B}]\op{B}^{n-1}                                                                        \\
		=                        & [\op{A},\op{B}^{n-2}]\op{B}^2+2[\op{A},\op{B}]\op{B}^{n-1}=\cdots=n[\op{A},\op{B}]\op{B}^{n-1}                                      \\
		\bullet~[\op{A},f(\op{B})] & =\sum_{n=0}^\infty\frac{f^{(n)}(0)}{n!}[\op{A},\op{B}^n]=\sum_{n=1}^\infty\frac{f^{(n)}(0)}{n!}[\op{A},\op{B}]n\op{B}^{n-1} \\
		                         & =\sum_{n=1}^\infty\frac{f^{(n)}(0)}{(n-1)!}\op{B}^{n-1}[\op{A},\op{B}]=f'(\op{B})[\op{A},\op{B}].
	\end{align*}
\end{theorem}
\subsection{Measurement}
The system in a state $\ket\psi$ which is normalized, and $\op{A}$ is any operator observable,
%$$\op{A}\ket{a_i}=a_i\ket{a_i}.$$
then $\ket\psi$ can be represented as
$$\ket\psi=\sum\ket{a_i}\ko3\brkt{a_i}\psi=\sum c_i\ket{a_i}$$
where $c_i=\brkt{a_i}\psi$ is the probability amplitude\index{probability amplitude} of getting $\ket{a_i}$ if measuring $\op{A}$.
\subsubsection{Probability and Expectation}
The possibility of getting $a_i$ is
$$\P(a_i)=\abs{c_i}^2=\bigabs{\brkt{a_i}\psi}^2.$$
The expectation\index{expectation} result when measuring $\op{A}$ is
$$\ave{\op{A}}=\sum \P(a_i)a_i=\sum\abs{c_i}^2a_i.$$
\begin{theorem}{}{}
	\centering$\spark{\ave{\op{A}}=\bra\psi\op{A}\ket\psi.}$
	\begin{flalign*}
		\begin{split}
			\text{Proof: }\bra\psi\op{A}\ket\psi&=\sum\bra\psi\op{A}\ket{a_i}\ko3\brkt{a_i}\psi=\sum\bra\psi a_i\ket{a_i}\ko3\brkt{a_i}\psi\\
			&=\sum a_i\brkt\psi{a_i}\brkt{a_i}\psi=\sum\abs{c_i}^2a_i,
		\end{split}&
	\end{flalign*}
\end{theorem}
For continuous case, the \textbf{probability density}\index{probability density} is
\begin{gather*}
	P(x)=\bigabs{\brkt x\psi}^2=\abs{\psi(x)}^2,\\
	\brkt\psi\psi=\int\brkt\psi{x}\ko3\brkt x\psi\d x=\int\abs{\psi(x)}^2\d x=\int P(x)\d x=1.
\end{gather*}
$P(x)\d x$ is the probability between $x$ and $x+\d x$.\\
And the average
$$\ave{\op{A}}=\bra\psi\op{A}\ket\psi=\int\bra\psi\op{A}\ket x\ko3\brkt x\psi\d x$$
\begin{example}{}{}
	The average of $x,p$ is
	$$\ave{\op{X}}=\int x\abs{\psi(x)}^2\d x,\quad\ave{\op{P}}=-\i\hbar\int\psi'(x) \psi\conj(x)\d x$$
	Warning: $\xcancel{\ave{\op{P}}=\int p(x)\abs{\psi(x)}^2\d x}$.
\end{example}

\subsubsection{Uncertainty}
If the measurements result in many values, then the deviation is $\Delta\op{A}=\op{A}-\ave{\op{A}}$
\begin{align*}
	\ave{\Delta\op{A}^2}= & \bigave{(\op{A}-\ave{\op{A}})^2}=\ave{\op{A}^2-2\op{A}\ave{\op{A}}+\ave{\op{A}}^2} \\
	=                  & \ave{\op{A}^2}-2\ave{\op{A}}\ko3\ave{\op{A}}+\ave{\op{A}}^2=\ave{\op{A}^2}-\ave{\op{A}}^2.
\end{align*}
Define the uncertainty\index{uncertainty} $\sigma_{\op{A}}^2:=\ave{\Delta\op{A}^2}$.\footnote{Textbooks tend to confuse $\Delta\op{A}$ and $\sigma_{\op{A}}$, it's understandable because $\Delta\op{A}$'s original definition in a single experiment doesn't matter.}
\begin{theorem}{Uncertainty Principle}{}
	lemma Schwarz
	\begin{equation}
		\brkt\alpha\alpha\ko3\brkt\beta\beta\geqslant\bigabs{\brkt\alpha\beta}^2.
	\end{equation}
		Proof: $\forall\lambda\in\RR,$
		\begin{align*}
			  & \bignorm{\ket\alpha+\lambda\ket\beta}^2=(\bra\alpha+\lambda\bra\beta)(\ket\alpha+\lambda\ket\beta)        \\
			= & \brkt\alpha\alpha+\lambda\bigl(\brkt\alpha\beta+\brkt\beta\alpha\bigr)+\lambda^2\brkt\beta\beta\geqslant 0,
		\end{align*}
		for $\lambda$, it is a quadratic inequality, so
		\begin{align*}
			\Delta & =\bigl(2\,\text{Re}\ko2\brkt\alpha\beta\bigr)^2-4\brkt\alpha\alpha\ko3\brkt\beta\beta \\
			       & =4\bigabs{\brkt\alpha\beta}^2-4\brkt\alpha\alpha\ko3\brkt\beta\beta\leqslant 0,
		\end{align*}
		that is what we need to proof.
	lemma
		If $\op{A}\adjoint=\op{A}$ Hermitian, $\ave{\op{A}}\in\RR$ for
		$$\bra\psi\op{A}\ket\psi\adjoint=\bra\psi\op{A}\adjoint\ket\psi=\bra\psi\op{A}\ket\psi.$$
		If $\op{A}\adjoint=-\op{A}$ anti-Hermitian, $\ave{\op{A}}\in \i\,\RR$ for
		$$\bra\psi\op{A}\ket\psi\adjoint=\bra\psi\op{A}\adjoint\ket\psi=-\bra\psi\op{A}\ket\psi.$$
	We take $\ket\alpha\rightarrow\Delta\op{A}\ket\psi,\ket\beta\rightarrow\Delta\op{B}\ket\psi$, from the Schwarz Lemma,\index{Schwarz Lemma}
	$$\sigma_{\op{A}}^2\sigma_{\op{B}}^2=\ave{\Delta\op{A}^2}\ko3\ave{\Delta\op{B}^2}\geqslant\bigabs{\ave{\Delta\op{A}\Delta\op{B}}}^2.$$
	Noticing that
	$$\ave{\Delta\op{A}\Delta\op{B}}=\ave{(\op{A}-\ave{\op{A}})(\op{B}-\ave{\op{B}})}\\
		=\ave{\op{A}\op{B}}-\ave{\op{A}}\ko3\ave{\op{B}}.$$
	Decompose $\op{A}\op{B}$,
	$$\op{A}\op{B}=\textstyle\frac12[\op{A},\op{B}]+\frac12\{\op{A},\op{B}\},$$
	$[\op{A},\op{B}]=\op{A}\op{B}-\op{B}\op{A}$ is anti-Hermitian, and $\{\op{A},\op{B}\}=\op{A}\op{B}+\op{B}\op{A}$ is Hermitian,
	$$\ave{\Delta\op{A}\Delta\op{B}}=\ave{\op{A}\op{B}}+\ave{\op{A}}\ko3\ave{\op{B}}=\underset{\text{Im-part}}{\underline{\textstyle\frac12\ave{[\op{A},\op{B}]}}}+\underset{\text{Re-part}}{\underline{\textstyle\frac12\ave{\{\op{A},\op{B}\}}-\ave{\op{A}}\ko3\ave{\op{B}}}}.$$
	Then
	$$\sigma_{\op{A}}\sigma_{\op{B}}\geqslant\bigabs{\ave{\Delta\op{A}\Delta\op{B}}}\geqslant\bigabs{\text{Im}\ko2\ave{\Delta\op{A}\Delta\op{B}}}=\textstyle\frac12\bigabs{\ave{[\op{A},\op{B}]}}.$$
	For $[\op{X},\op{P}]=\i\hbar$, we conduct the \textbf{Uncertainty Principle}\index{Uncertainty Principle}
	$$\spark{\sigma_{\op{X}}\sigma_{\op{P}}\geqslant\frac{\hbar}2,}$$
	which means we can't precisely measure $\op{X}$ and $\op{P}$ simultaneously.
\end{theorem}
\subsection{Schrödinger Equation}
The Schrödinger Equation\index{Schrödinger Equation} is
\spark{\begin{align}\label{vectdse}
		\i\hbar\dv{\ket\psi}t=\op{H}\ket\psi,
	\end{align}}
where Hamiltonian $\op{H}=\op{K}+V(\op{X})$.
\subsubsection{Time Dependent Schrödinger Equation}
Left multiply Eqn.(\ref{vectdse}) by $\bra x$, we get\index{Time Dependent Schrödinger Equation}
\begin{align}\label{tdse}
	\i\hbar\pv{\psi(x,t)}t=-\frac{\hbar^2}{2m}\pv[2]{\psi(x,t)}x+V(x)\psi(x,t),
\end{align}
\begin{definition}{Probability Current}{}
	Let $P_{[a,b]}(t)$ be the probability of finding the particle in $[a,b]$ at time $t$.
	$$P_{[a,b]}(t)=\P(a\leqslant X\leqslant b)=\int_a^b\abs{\psi(x,t)}^2\d x.$$
	From Sch-Eqn.(\ref{tdse}): $\pv\psi{t}=\frac{\i\hbar}{2m}\pv[2]\psi{x}+\frac1{\i\hbar}V\psi$,
	\begin{align*}
		\begin{aligned}
			\dv{P_{[a,b]}}t & =\dd t\int_a^b\psi\conj\psi\d x=\int_a^b\Bigp{\pv{\psi\conj}t\psi+\psi\conj\pv\psi{t}}\d x                                                                                        \\
			        & =\int_a^b\biggb{\Bigp{-\frac{\i\hbar}{2m}\pv[2]{\psi\conj}x-\cancel{\ko1\frac1{\i\hbar}V}\psi\conj}\psi+\psi\conj\Bigp{\frac{\i\hbar}{2m}\pv[2]\psi{x}+\ko1\cancel{\frac1{\i\hbar}V}\psi}}\d x
		\end{aligned} \\
		\begin{aligned}
			 & =\frac{\i\hbar}{2m}\int_a^b\ko2\Bigp{\psi\conj\pv[2]\psi{x}-\psi\pv[2]{\psi\conj}x}\d x=\frac{\i\hbar}{2m}\int_a^b\pp x\Bigp{\psi\conj\pv\psi{x}-\psi\pv{\psi\conj}x}\d x \\
			 & =\frac{\i\hbar}{2m}\Bigb{\psi\conj\pv\psi{x}-\psi\pv{\psi\conj}x}_a^b=:j(a,t)-j(b,t),
		\end{aligned}
	\end{align*}
	where $j:=\frac{\i\hbar}{2m}\Bigp{\psi\pv{\psi\conj}x-\psi\conj\pv{\psi}x}$ is the \textbf{probability current}.\index{probability current}
\end{definition}
Directly solving Eqn.(\ref{tdse}) is difficult, we need the eigenfunctions.
%in 3-D    $$\i\hbar\pv{\psi}t=-\frac{\hbar^2}{2m}\laplacian\psi+V\psi$$
\subsubsection{Time Independent Schrödinger Equation}
If we use $\ket E$ base\index{base}, which is the eigenvector of $\op{H}$, i.e.
\begin{align}\label{vectise}
	\op{H}\ket E=E\ket E,
\end{align}\index{Time Independent Schrödinger Equation}
and left multiply Eqn.(\ref{vectdse}) by $\bra E$,
$$\bra E\op{H}\ket\psi=\i\hbar\dv{\brkt E\psi}t=E\brkt E\psi.$$
Define $\zeta(t):=\brkt E\psi$ as a function of $t$, then
$$\i\hbar\dv{\zeta(t)}t=E\zeta(t),$$
which is easy to solve and the solution is
$$\zeta(t)=\zeta(0)\e^{E/\i\hbar\,t}.$$
Because $\ket\psi=\sum\ket{E_n}\ko3\brkt{E_n}\psi=\sum\zeta_n(t)\ket{E_n}$,
$$\ket\psi=\sum\zeta_n(0)\e^{-\i\omega_nt}\ket{E_n},\quad\omega_n=\frac{E_n}\hbar,$$
Define $\psi(x):=\brkt xE$, left multiply Eqn.(\ref{vectise}) by $\bra x$,
\spark{\begin{align}\label{tise}
		-\frac{\hbar^2}{2m}\dv[2]{\psi(x)}x+V(x)\psi(x)=E\psi(x).
	\end{align}}\su{20}
\paragraph{Clarify}Here $\psi(x)$ is \textbf{different} from $\psi(x,t)$ in Eqn.(\ref{tdse}). $\psi(x,t)$ is the wave function; $\psi(x)$ is the eigenfunction of Eqn.(\ref{tise}), and it's independent of $t$.
\paragraph{Link}Taking different $E_n$, we get a series of $\psi_n(x)$ by sloving Eqn.(\ref{tise}), $\ket{E_n}$ is the base in space so
$$\psi(x,0)=\sum c_n\psi_n(x),$$
then as $t$ evolves,
$$\psi(x,t)=\sum c_n\e^{-\i\omega_nt}\psi_n(x),\quad\omega_n=\frac{E_n}\hbar.$$

\subsubsection{Ehrenfest Theorem\index{Ehrenfest Theorem}}
\begin{theorem}{Ehrenfest}{}
	$$\spark{\dv{\ave{\op{A}}}t=\frac 1{\i\hbar}\ave{[\op{A},\op{H}]}.}$$
	\begin{flalign*}
		\begin{split}
			\text{Proof: }\dv{\ave{\op{A}}}t=\dv{\bra\psi\op{A}\ket\psi}t=\dv{\bra\psi}t\op{A}\ket\psi+\bra\psi\op{A}\dv{\ket\psi}t.
		\end{split}\footnotemark[1]&
	\end{flalign*}\footnotetext[1]{Most operators are independent of time, i.e. $\partial\op{A}/\partial t\equiv 0.$}
	From Sch-Eqn.(\ref{vectdse}): $\dv{\ket\psi}t=\frac1{\i\hbar}\op{H}\ket\psi,\dv{\bra\psi}t=-\frac1{\i\hbar}\bra\psi\op{H}$, hence,
	%$$\dv{\bra\psi}t=\Bigl(\dv{\ket\psi}t\Bigr)\adjoint=\Bigl(\frac1{\i\hbar}\op{H}\ket\psi\Bigr)\adjoint=-\frac1{\i\hbar}\bra\psi\op{H},$$
	$$\dv{\ave{\op{A}}}t=\Bigl(-\frac1{\i\hbar}\bra\psi\op{H}\Bigr)\op{A}\ket\psi+\bra\psi\op{A}\Bigl(\frac1{\i\hbar}\op{H}\ket\psi\Bigr)=\frac 1{\i\hbar}\ave{[\op{A},\op{H}]}.$$
	%$$=-\frac1{\i\hbar}\bra\psi\op{H}\op{A}\ket\psi+\frac1{\i\hbar}\bra\psi\op{A}\op{H}\ket\psi$$
\end{theorem}
\begin{example}{$\op{A}=\op{X}$}{}
	$$\spark{\dv{\ave{\op{X}}}t=\frac1{\i\hbar}\ave{[\op{X},\op{H}]}=\frac{\ave{\op{P}}}m.}$$
	For
	$$[\op{X},\op{H}]=\frac1{2m}[\op{X},\op{P}^2]+[\op{X},V(\op{X})]=\frac{\op{P}}m[\op{X},\op{P}]+0=\frac{\i\hbar}m\op{P}.$$
	This makes perfect sense because in classic
	$$v=\dv xt=\frac pm.$$
\end{example}
\begin{example}{$\op{A}=\op{P}$}{}
	$$\spark{\dv{\ave{\op{P}}}t=\frac1{\i\hbar}\ave{[\op{P},\op{H}]}=-\ave{V'(\op{X})}.}$$
	For\su5
	$$[\op{P},\op{H}]=\frac1{2m}[\op{P},\op{P}^2]+[\op{P},V(\op{X})]=0+V'(\op{X})[\op{P},\op{X}]=-\i\hbar V'(\op{X}).$$
	This alse makes sense,
	$$F=\dv pt=-\dv{V(x)}x.$$
	In 3-D space,
	$$\dv{\ave{\op{P}}}t=-\ave{\nabla V},$$
	and for the angular momentum, like $\bm\tau=\dv{\bm L}t=\bm r\times\bm F,$
	$$\dv{\ave{\op{L}}}t=\ave{-\bm r\times\nabla V(\bm r)}$$
\end{example}
\begin{theorem}{Time-Energy Uncertainty Principle}{}
	Let's go back to the Uncertainty Principle\index{Uncertainty Principle}
	$$\sigma_{\op{A}}\sigma_{\op{B}}\geqslant\frac12\bigabs{\ave{[\op{A},\op{B}]}},$$
	when $\op{B}\equiv\op{H}$, from Ehrenfest Theorem: $\ave{[\op{A},\op{H}]}=\i\hbar~\d\ko2\ave{\op{A}}\ko3/\d t$
	$$\sigma_{\op{A}}\sigma_{\op{H}}\geqslant\frac12\bigabs{\ave{[\op{A},\op{H}]}}=\frac{\hbar}2\Bigabs{\dv{\ko2\ave{\op{A}}}t},$$
	when $\op{A}\equiv\op{T}$, which is the time operator
	$$\sigma_{\op{T}}\sigma_{\op{H}}\geqslant\frac{\hbar}2.$$
\end{theorem}

\subsubsection{Example: Spin-1/2 System}

\subsection{Harmonic Oscillator}
$V(x)=\frac12kx^2$, the Time Independent Schrödinger Equation is
$$-\frac{\hbar^2}{2m}\dv[2]\psi{x}+\frac12kx^2\psi=E\psi.$$
$\omega^2:=\frac km,\xi:=\sqrt{\frac{m\omega}\hbar}x,K:=\frac{2E}{\hbar\omega},$ the equation becomes
\begin{align}\label{hereqn}
	\dv[2]\psi\xi+(K-\xi^2)\psi=0,
\end{align}
which is Hermite Equation.\index{Hermite Equation}\\
Considering the asymptote behavior: when $\xi\to\infty$, Eqn.(\ref{hereqn}) approach
$$\dv[2]\psi\xi-\xi^2\psi=0,$$
thus when $\xi\to\infty$, $\psi\to A\e^{-\xi^2/2}$ ($B\e^{\xi^2/2}$ diverges unless $B\equiv 0$).\\
Guess $A=h(\xi)$, expand it
\begin{center}
	$h(\xi)=a_0+a_1\xi+a_2\xi^2+\cdots=\sum_{i=0}^\infty a_i\xi^i,$
\end{center}
because $V(x)$ is even, $h(\xi)$ can be either odd or even, i.e.
\begin{center}
	$h(\xi)=\sum_{m=0}^\infty a_j\xi^j,\quad j\equiv 2m~{\rm xor}~j\equiv 2m+1$.\footnote{The xor (exclusive or) means either one, but not both. Its symbol $\oplus$ is too ugly to use.}
\end{center}
Substitute into the original Eqn.(\ref{hereqn}),
\begin{gather*}
	\begin{aligned}
		\dv\psi\xi  & =(h'-\xi h)\e^{-\xi^2/2},            \\
		\dv[2]\psi\xi & =[h''-2\xi h'+(\xi^2-1)h]\e^{-\xi^2/2},
	\end{aligned}\\
	\dv[2]\psi\xi+(K-\xi^2)\psi=[h''-2\xi h'+(K-1)h]\e^{-\xi^2/2}.
\end{gather*}
Thus $h''-2\xi h'+(K-1)h=0$:
\begin{gather*}
	\sum_{m=0}^\infty j(j-1)a_j\xi^{j-2}-2\sum_{m=0}^\infty ja_j\xi^j+(K-1)\sum_{m=0}^\infty a_j\xi^j=0,\\
	\Rightarrow\quad a_{j+2}=\frac{2j+1-K}{(j+2)(j+1)}a_j.
\end{gather*}
When $j\to\infty$,
$$\frac{a_{j+2}\xi^{j+2}}{a_j\xi^j}\to\frac{\xi^2}m.$$
While $\e^{\xi^2}=\sum_{m=0}^\infty\frac{\xi^{2m}}{m!}$, i.e.\su{10}
$$h(\xi)\to \e^{\xi^2},\quad\psi\to \e^{\xi^2/2}~{\rm(diverges)}.$$
The only way out of the dilemma is $a_j=0$ when $j\geqslant n$, i.e. $K=2n+1,$
$$\spark{E_n=\frac{\hbar\omega}2K=\Bigl(n+\frac12\Bigr)\hbar\omega,\quad n=0,1,2,\ldots.}$$
For the certain $n$, $\psi_n=h_n(\xi)\e^{-\xi^2/2}$, then work out the coefficients through the recursion
$$a_{j-2}=\frac{j(j-1)}{2(j-n-2)}a_j,\quad j=n,n-2,\ldots,$$
and the normalization $\brkt{\psi_n}{\psi_n}=1$.
\begin{example}{$n=0$}{}
	$h(\xi)=a_0$, $\psi_0=a_0\e^{-\xi^2/2}$,
	$$\intR \ko5\psi_0^2\d x=a_0^2\intR \ko5\e^{-m\omega x^2/\hbar}\d x=a_0^2\sqrt{\frac{\pi\hbar}{m\omega}}=1.$$
	Thus
	$\psi_0=\sqrt[4]{\frac{m\omega}{\pi\hbar}}\e^{-\xi^2/2}.$
\end{example}
The general wave function is
$$\spark{\psi_n=\sqrt[4]{\frac{m\omega}{\pi\hbar}}\frac1{\sqrt{2^nn!}}H_n(\xi)\e^{-\xi^2/2},\quad\xi=\sqrt{\frac{m\omega}\hbar}x.}$$
where the Hermite Polynomial \index{Hermite Polynomial}
$$H_n(x)=(-1)^n\e^{x^2}\dd[n]{x}\e^{-x^2}.$$
\begin{example}{Table of Hermite}{}
	The first few items are %($\psi \e^{\xi^2/2}$)
	\begin{align*}
		H_0 & =1,  & H_2 & =4x^2-2,   & H_4 & =16x^4-48x^2+12,    \\
		H_1 & =2x, & H_3 & =8x^3-12x, & H_5 & =32x^5-160x^3+120x.
	\end{align*}
	\iffalse
		\begin{align*}
			\psi_0 & \propto \e^{-\xi^2/2}            & \psi_3 & \propto\frac1{\sqrt3}(2\xi^3-3\xi),
			\psi_1 & \propto\sqrt2\xi \e^{-\xi^2/2}   & \psi_4 & \propto\frac1{2\sqrt6}(4\xi^4-12\xi^3+3),     \\
			\psi_2 & \propto\frac1{\sqrt2}(2\xi^2-1) & \psi_6 & \propto\frac1{2\sqrt{30}}(4\xi^5-20\xi^3+15).
		\end{align*}
	\fi
\end{example}

\subsection{Hydrogen Atom}
For a system consists of a proton $\proton$ and an electron $\electron$, the distance between is $r$. The Hamiltonian in $\ket x$ base is
$$\op{H}=-\frac{\hbar^2}{2m_{\proton}}\laplacian_{\proton}-\frac{\hbar^2}{2m_{\electron}}\laplacian_{\electron}+V(r),$$
%where $\laplacian_N$ only act on the $R_N$ part, and $V(r)=-\frac{e^2}{4\pi\varepsilon_0r}$\\
Decompose $\op{H}$ into the free-particle motion of the total mass, and relative motion of reduced mass.\\
For the center of mass part, $M=m_{\proton}+m_{\electron}$, $R_{\CM}=\frac{m_{\proton}R_{\proton}+m_{\electron}R_{\electron}}{m_{\proton}+m_{\electron}}$; for the reduced mass part, $m=\frac{m_{\proton}m_{\electron}}{m_{\proton}+m_{\electron}}$, $r=R_{\proton}-R_{\electron}$.
$$\op{H}_{\CM}=-\frac{\hbar^2}{2M}\nabla_{\CM}^2,\qquad\op{H}_m=-\frac{\hbar^2}{2m}\nabla_r^2+V(r).$$
We know how to solve the free particle and here we shall only concentrate on the relative motion $\psi(r,\theta,\phi)$:
\begin{align}\label{s-eqn}
	-\frac{\hbar^2}{2m}\laplacian\psi+V\psi=E\psi.
\end{align}
\paragraph{Separation of Variables}
Since the potential $V(r)$ only depends on distance, not on direction. It has the spherical symmetry, %In Cartesian coordinate,$$\laplacian=\pp[2]x+\pp[2]y+\pp[2]z,$$
in spherical coordinate
$$\laplacian=\frac1{r^2}\pp r\Bigp{r^2\pp r}+\frac1{r^2\sin\theta}\pp \theta\Bigp{\sin\theta\pp \theta}+\frac1{r^2\sin^2\theta}\pp[2]\phi.$$
Because the three variables $r,\theta,\phi$ are independent, the wave function can be decomposed, i.e.%\footnote[1]{I've gotten a fantastic proof, pityfully the space here is too small to write.} 
$$\psi(r,\theta,\phi)=R(r)\varTheta(\theta)\varPhi(\phi),$$
where the $R(r)$ is the radial part, and $Y(\theta,\phi)=\varTheta(\theta)\varPhi(\phi)$ is the angular part.
Then substitute into the Eqn.(\ref{s-eqn}),
\begin{gather*}
	-\frac{\hbar^2}{2mr^2}\biggl[\dd r\ko2\Bigl(r^2\dv Rr\Bigr)\ko2+\ko2\frac1{\sin\theta}\pp \theta\ko1\Bigl(\sin\theta\pv Y\theta\Bigr)\ko2+\ko2\frac1{\sin^2\theta}\pv[2] Y\phi\biggr]+(V-E)RY=0,\\
	\frac1R\dd r\Bigl(r^2\dv Rr\Bigr)-\frac{2mr^2}{\hbar^2}(V-E)=-\frac1Y\Lambda^2Y,
\end{gather*}
Thus $\mathrm{LHS}(r)=\mathrm{RHS}(\theta,\phi)=J$ (constant).
\paragraph{~}The Legendrian $\Lambda^2$\index{Legendrian $\Lambda^2$} have been mentioned in angular momentum, and $Y$ is the eigenfunction of $\op{L}^2$
$$\op{L}^2Y=-\hbar^2\Lambda^2Y=J\hbar^2Y.$$
\subsubsection{Solution of Legendrian}
$Y(\theta,\phi)=\varTheta(\theta)\varPhi(\phi)$, separate variables,
\begin{gather*}
	-\frac1{\varTheta\varPhi}\biggl[\frac1{\sin\theta}\dd \theta\Bigl(\sin\theta\dv\varTheta\theta\Bigr)+\frac1{\sin^2\theta}\dv[2]\varPhi\phi\biggr]=J,\\
	\frac{\sin\theta}\varTheta\dd \theta\Bigl(\sin\theta\dv\varTheta\theta\Bigr)+J\sin^2\theta=-\frac1\varPhi\dv[2]\varPhi\phi.
\end{gather*}
$\Rightarrow\quad \mathrm{LHS}(\theta)=\mathrm{RHS}(\phi)=m^2$ (constant).
\paragraph{Solving $\varPhi$ in the RHS}
$$\dv[2]\varPhi\phi+m^2\varPhi=0,\quad\Rightarrow\quad\varPhi(\phi)=\e^{\i m\phi}.$$
As $\varPhi(\phi+2\pi)=\varPhi(\phi)$, $m=0,\pm1,\pm2,\ldots;\varPhi$ is the eigenfunction of $\op{L}_z$
$$\op{L}_z\varPhi=-\i\hbar\pv\varPhi\phi=m\hbar\,\varPhi.$$
\paragraph{Solving $\varTheta$ in the LHS}\ko7\footnote{In the LHS, $m$'s sign dosen't really matter, thus we take $m$ positive.}
$$\frac{\sin\theta}\varTheta\dd \theta\Bigp{\sin\theta\dv\varTheta\theta}+J\sin^2\theta=m^2.$$
Let $x=\cos\theta,y=\varTheta(\theta)$, then
$$(1-x^2)\dv[2] yx-2x\dv yx+\Bigp{J-\frac{m^2}{1-x^2}}y=0.$$
Guess $y=(1-x^2)^{m/2}v$,
\begin{gather*}
	\dv yx=(1-x^2)^{m/2}\Bigp{v'-\frac{mx}{1-x^2}v},\\
	\dv[2] yx=(1-x^2)^{m/2}\Bigb{v''-\frac{2mx}{1-x^2}v'+\frac{m(m-1)x^2-m}{(1-x^2)^2}v}.
\end{gather*}
Thus $(1-x^2)v''-2(m+1)xv'+[J-m(m+1)]v=0.$
\begin{gather*}
	\sum t(t-1)c_tx^{t-2}-\sum t(t-1)c_tx^t\\
	-2(m+1)\sum tc_tx^t+[J-m(m+1)]\sum c_tx^t=0,\\
	\Rightarrow\quad c_{t+2}=\frac{(t+m+1)(t+m)-J}{t(t+1)}c_t.
\end{gather*}
To converge, $J=\ell(\ell+1)$, and $\ell=t_0+m=0,1,2,3,\ldots,$ (s, p, d, f, $\ldots$)
$$\varTheta(\theta)=AP_\ell^m(\cos\theta),$$
where Legendre Function $P_\ell^m(x)$\index{Legendre Function}
$$P_\ell^m(x)=\frac{(1-x^2)^{m/2}}{2^\ell\ell!}\frac{\d^{\ell+m}}{\d x^{\ell+m}}(x^2-1)^\ell,$$
$$Y_\ell^m=\pm\sqrt{\frac{2\ell+1}{4\pi}\frac{(\ell-m)!}{(\ell+m)!}}P_\ell^m(\cos\theta)\e^{\i m\phi},$$
where $-$ is taken only when $m=1,3,5,\ldots.$

\subsubsection{Solution of Radial Part}
$$\frac1R\dd r\Bigp{r^2\dv Rr}-\frac{2mr^2}{\hbar^2}(V-E)=\ell(\ell+1),$$
where $V=-\frac{e^2}{4\pi\varepsilon_0r}$.\\
Noticing
$$\dd r\Bigp{r^2\dv Rr}=2r\dv Rr+r^2\dv[2] Rr=r\dv[2]{rR}r.$$
To simplify, define $u:=rR$, $k=\sqrt{-\frac{2mE}{\hbar^2}}$, $\xi:=kr$, $N=\frac{me^2}{2\pi\varepsilon_0\hbar^2k}$,
\begin{align}\label{radeqn}
	\dv[2] u\xi=\Bigl[1-\frac N\xi+\frac{\ell(\ell+1)}{\xi^2}\Bigr]u.
\end{align}
Asymptote behavior
\begin{align*}
	\xi & \to+\infty, & \dv[2] u\xi & =u                           & \Rightarrow\quad u & \to A\e^{-\xi},(B\e^\xi{\rm~diverges})           \\
	\xi & \to0,       & \dv[2] u\xi & =\frac{\ell(\ell+1)}{\xi^2}u & \Rightarrow\quad u & \to C\xi^{\ell+1},(D\xi^{-\ell}{\rm~diverges})
\end{align*}
therefore, $u=v(\xi)\xi^{\ell+1}\e^{-\xi}.$
\begin{gather*}
	\dv u\xi=[v'\xi+v(\ell+1)-v\xi]\xi^\ell \e^{-\xi},\\
	\dv[2] u\xi=[v''\xi^2+2v'(\ell+1)\xi-2v'\xi^2+v(\ell+1)\ell-2v(\ell+1)\xi+v\xi^2]\xi^{\ell-1}\e^{-\xi}\\
	\Bigl[1-\frac N\xi+\frac{\ell(\ell+1)}{\xi^2}\Bigr]u=[\xi^2-N\xi+\ell(\ell+1)]v\xi^{\ell-1}\e^{-\xi}.
\end{gather*}
Then $\xi v''+2(\ell+1-\xi)v'+[N-2(\ell+1)]v=0.$
\begin{gather*}
	\sum t(t-1)c_t\xi^{t-1}+2(\ell+1)\sum tc_t\xi^{t-1}\qquad\\
	\qquad-2\sum tc_t\xi^t+[N-2(\ell+1)]\sum c_t\xi^t=0.\\
	\Rightarrow\quad c_{t+1}=\frac{2(t+\ell+1)-N}{(t+1)(t+2\ell+2)}c_t.
\end{gather*}
To converge, $N=2n$, and $n=t_0+\ell+1$.
\begin{gather*}
	N=2n=\frac{me^2}{2\pi\varepsilon_0\hbar^2}\frac1k,\quad
	k=\frac1n\frac{me^2}{4\pi\varepsilon_0\hbar^2},\\
	E=-\frac{\hbar^2k^2}{2m}=-\frac1{n^2}\frac m{2\hbar^2}\Bigl(\frac{e^2}{4\pi\varepsilon_0}\Bigr)^2.
\end{gather*}
Define the reduced Bohr radius\footnote{The original Bohr radius uses $m_{\electron}$ in the mass part, the reduced mass $m$ (or $\mu$) $\doteq 0.999\,m_{\electron}$.}
$$a=\frac{4\pi\varepsilon_0\hbar^2}{me^2}=\qty{0.53e-10}{m},$$
and the ground energy at $n=1$,
$$E_1=-\frac m{2\hbar^2}\Bigl(\frac{e^2}{4\pi\varepsilon_0}\Bigr)^2=\qty{-13.6}{eV}.$$
For the certain $n,\ell,k=\frac1{na}$, $\xi=kr$
$$R=\frac ur=v(\xi)\xi^\ell \e^{-\xi},~(k{\rm~in~the}~v)$$
then work out the coefficients through the recursion
$$c_{j+1}=\frac{2(j+\ell+1-n)}{(j+1)(j+2\ell+2)}c_j,\quad j=0,1,\ldots,n-\ell-2,$$
and the normalization.
\begin{example}{1s orbit}{}
	$n=1,\ell=0,m=0,R=c_0\e^{-\xi}=c_0\e^{-r/a}$,
	$$\intRplus R^2r^2\d r=c_0^2\intRplus r^2\e^{-2r/a}d r=\frac{a^3c_0^2}4=1.$$
	Thus
	$$R_{10}=\frac2{\sqrt{a^3}}\e^{-r/a},$$
	While $Y_0^0=\frac1{2\sqrt\pi}$, $\psi_{100}=R_{10}Y_0^0=\frac1{\sqrt{\pi a^3}}\e^{r/a}$.
\end{example}
Define $\rho=2\xi=\frac{2r}{na},p=2\ell+1,q=n-\ell-1$, then
$$\rho v''+(p+1-\rho)v'+qv=0,\quad\Rightarrow\quad v\propto L_q^p(\rho).$$
$$R_{n\ell}=\sqrt{\Bigp{\frac2{na}}^3\frac{(n-\ell-1)!}{2n(n+l)!}}\rho^\ell \e^{-\rho/2}L_q^p(\rho),$$
where the Laguerre Function $L_q^p(x)$\index{Laguerre Function}
$$L_q^p(x)=(-1)^pL_{p+q}^{(p)}(x),$$
and the Laguerre Polynomial $L_q(x)$\index{Laguerre Polynomial}
$$L_q(x)=\frac{\e^x}{q!}\dd[q]{x}\frac{x^q}{\e^x}.$$

\paragraph{The meaning of \textit{n,l,m}}
\begin{itemize}
	\item $n$ is the Principle Quantum Number: $\op{H}\ket\psi=E_1n^{-2}\ket\psi.$
	\item $\ell$ is the Azzimuthal Quantum Number: $\op{L}^2\ket\psi=\ell(\ell+1)\hbar^2\ket\psi.$
	\item $m$ is the Magnetic Quantum Number: $\op{L}_z\ket\psi=m\hbar\ket\psi.$
\end{itemize}
The values are quantized: $n=1,2,3,\ldots;\ell=0,1,\ldots,n-1;m=0,\pm 1,\ldots,\pm\ell$.
\paragraph{More} $n,\ell,m$ is still not enough, the spin $\op{S}$ should also be taken into account, i.e.
$$\ket\psi=\ket{n,\ell,m}\otimes\ket{s,s_z}.$$
$\otimes$ is the tensor product, meaning the value of $s,s_z$ is independent of $n,\ell,m$.

\subsection{Nabla}\index{nabla $\nabla$}
\subsubsection{Definition}
\paragraph{Introduction}We could use linear function to approximate a function near a certain point, that is
\begin{align*}
	f(x)\sim   & f(x_0)+f'(x_0)(x-x_0)                                                                \\
	=          & f(x_0)+f'(x_0)\Delta x,                                                              \\
	f(x,y)\sim & f(x_0,y_0)+f'_x(x_0,y_0)(x-x_0)+f'_y(x_0,y_0)(y-y_0)                                 \\
	=          & f(x_0,y_0)+\Bigl[\pv fx(x_0,y_0),\pv fy(x_0,y_0)\Bigr]\cdot[\Delta x,\Delta y],
\end{align*}
then $f(x,y,z)$ at $P_0(x_0,y_0,z_0)$
$$\nabla f(P_0):=\Bigl[\pv fx(P_0),\pv fy(P_0),\pv fz(P_0)\Bigr]=\Bigl[\pv fx,\pv fy,\pv fz\Bigr]_{P_0}$$
$$f(P)\sim f(P_0)+\nabla f(P_0)\cdot\Delta P.$$
\paragraph{Gradient} In Cartesian coordinates,\index{gradient}
$$\nabla f=\pv fx\uvec i+\pv fy\uvec j+\pv fz\uvec k.$$
We take the notation nabla $\nabla$
$$\nabla=\pp x\uvec i+\pp y\uvec j+\pp z\uvec k,$$
which is very useful later.
\paragraph{Divergence} The flux $\Phi$ of $\bm F$ through a surface $S$.\index{divergence}
$$\Phi=\int_S\bm F\cdot\d\bm S.$$
If closed surface $S=\partial V$,
$$\Phi=\oint_{\partial V}\ko5\bm F\cdot\d\bm S=\sum_{i=1}^N\frac1{V_i}\oint_{\partial V_i}\ko5\bm F\cdot\d\bm S\,V_i.$$
Define divergence
$$\text{div}\bm F:=\lim_{V\to0}\frac1V\oint_{\partial V}\ko5\bm F\cdot\d\bm S,$$
then we conduct the Gauss's law
$$\Phi=\oint_{\partial V}\ko5\bm F\cdot\d\bm S=\int_V\text{div}\bm F\,\d V.$$
Take $V$ as a cube origin at $(x,y,z)$ with a delta $(\Delta x,\Delta y,\Delta z)$, thus
$$\Delta V=\Delta x\,\Delta y\,\Delta z$$
In the $z$ direction,
\begin{align*}
	\int F_z\,\d S_z & = F_z(x,y,z+\Delta z)\Delta x\Delta y- F_z(x,y,z)\Delta x\Delta y           \\
	                 & =\frac{ F_z(x,y,z+\Delta z)- F_z(x,y,z)}{\Delta z}\Delta x\Delta y\Delta z.
\end{align*}
Therefore,
$$\text{div}\bm F=\pv{ F_x}x+\pv{ F_y}y+\pv{ F_z}z.$$
Noticing that this formally fit
$$\div\bm F=\Bigl[\pp x,\pp y,\pp z\Bigr]\cdot[F_x, F_y, F_z]=\pp x F_x+\pp y F_y+\pp z F_z,$$
we can use the notation:
$$\div\bm F=\text{div}\bm F=\pv{ F_x}x+\pv{ F_y}y+\pv{ F_z}z.$$
\paragraph{Curl} Path integral\index{curl}
$$\oint_{\partial S}\ko5\bm F\cdot\d\bm\ell=\sum_{i=1}^N\frac1{S_i}\oint_{\partial S_i}\ko5\bm F\cdot\d\bm\ell\,S_i.$$
Define curl, whose projection along the unit vector $\uvec{n}$ direction is
$$\text{curl}\bm F\cdot\uvec{n}:=\lim_{S\to0}\frac1S\oint_{\partial S}\ko5\bm F\cdot\d\bm\ell,$$
then we conduct the Stokes's law
$$\oint_{\partial S}\ko5\bm F\cdot\d\bm\ell=\int_S\text{curl}\bm F\cdot\d\bm S.$$
As $\uvec{n}=\uvec{k}$, take $S_k$ as a square origin at $(x,y,z)$ with a delta $(\Delta x,\Delta y,0)$, thus
\begin{gather*}
	\Delta S_k=\Delta x\Delta y.\\
	%&\oint_{\partial S_k}\ko5\bm F\cdot\d\bm\ell\\=
	F_x(x,y,z)\Delta x+F_y(x+\Delta x,y,z)\Delta y-F_x(x,y+\Delta y,z)\Delta x-F_y(x,y,z)\Delta y\\
	=\Bigb{\frac{F_x(x,y,z)-F_x(x,y+\Delta y,z)}{\Delta y}+\frac{F_y(x+\Delta x,y,z)-F_y(x,y,z)}{\Delta x}}\Delta x\Delta y.
\end{gather*}
Therefore,
$$(\text{curl}\bm F)_z=\pv{F_y}x-\pv{F_x}y.$$
That is,
$$\text{curl}\bm F=\Bigb{\pv{F_z}y-\pv{F_y}z,\pv{F_x}z-\pv{F_z}x,\pv{F_y}x-\pv{F_x}y}.$$
Noticing that this formally fit
\begin{align*}
	\curl\bm F=
	\begin{vmatrix}

\subsubsection{Nabla Notation in Coordinate Transformation}\index{coordinate transformation}
In Cartesian coordinates, bases $\{\uvec i,\uvec j,\uvec k\}$,
$$\bm r=x_1\uvec i+x_2\uvec j+x_3\uvec k.$$
In another orthogonal normalized bases $\{\uvec e_1,\uvec e_2,\uvec e_3\}$,
$$\bm r=\xi_1\uvec e_1+\xi_2\uvec e_2+\xi_3\uvec e_3,$$
We have the mapping $\mathcal T$:
$$\mathcal T:(\xi_1,\xi_2,\xi_3)\rightarrow(x_1,x_2,x_3),$$
i.e. $x_i=x_i(\xi_1,\xi_2,\xi_3)$, and
$$\d x_i=\pv{x_i}{\xi_1}\d\xi_1+\pv{x_i}{\xi_2}\d\xi_2+\pv{x_i}{\xi_3}\d\xi_3,$$
compose as $\d\bm r=\d x_1\uvec i+\d x_2\uvec j+\d x_3\uvec k$,\footnote{Warning: $\partial\bm r/\partial\xi_i\neq\partial r/\partial\xi_i\uvec r$, i.e. $\partial\bm r/\partial\xi_i$ isn't along $\uvec r$.}   %\partial x_1/\partial\xi_i\uvec i+\partial x_2/\partial\xi_i\uvec j+\partial x_3/\partial\xi_i\uvec k
$$\d\bm r=\pv{\bm r}{\xi_1}\d\xi_1+\pv{\bm r}{\xi_2}\d\xi_2+\pv{\bm r}{\xi_3}\d\xi_3.$$
Decompose the small displacement $\d\bm r$ along $\uvec e_1,\uvec e_2,\uvec e_3$ directions:
$$\d\bm r=\d\bm\ell_1+\d\bm\ell_2+\d\bm\ell_3.$$
Noticing that $\d\ell_i\neq\d\xi_i$\footnote{If so, $\d\bm r=\d\xi_1\uvec e_1+\d\xi_2\uvec e_2+\d\xi_3\uvec e_3$, the new coordinate is just a rotated Cartesian, the form won't change.}, we use the Lame coefficient: $\d\bm\ell_i=H_i\d\xi_i\uvec e_i$.
$$\d\bm r=H_1\d\xi_1\uvec e_1+H_2\d\xi_2\uvec e_2+H_3\d\xi_3\uvec e_3.$$
Thus
$$\pv{\bm r}{\xi_i}=\pv{x_1}{\xi_i}\uvec i+\pv{x_2}{\xi_i}\uvec j+\pv{x_3}{\xi_i}\uvec k=H_i\uvec e_i,$$
we can calculate $H_i$
$$H_i=\Bignorm{\pv{\bm r}{\xi_i}}=\sqrt{\Bigl(\pv{x_1}{\xi_i}\Bigr)^2+\Bigl(\pv{x_2}{\xi_i}\Bigr)^2+\Bigl(\pv{x_3}{\xi_i}\Bigr)^2}$$

\paragraph{Gradient} Follow the definition\index{gradient}
$$\nabla f:=\sum_{i=1}^3\pv f{\bm\ell_i}=\sum_{i=1}^3\uvec e_i\frac{\partial f}{H_i\,\partial\xi_i}.$$
We take the notation:
$$\nabla\equiv\sum_{i=1}^3\uvec e_i\frac\partial{H_i\partial\xi_i}.$$
\paragraph{Divergence} Follow the definition\index{devergence}
$$\div\bm F:=\lim_{V\to0}\frac1{V}\oint_{\partial V}\ko5\bm F\cdot\d\bm S.$$
Taking $V$ as a cube origin at $(\xi_1,\xi_2,\xi_3)$ with a delta $(\d\ell_1,\d\ell_2,\d\ell_3)$, thus
\begin{gather*}
	\d V=\d\ell_1\wedge\d\ell_2\wedge\d\ell_3=H_1H_2H_3\d\xi_1\d\xi_2\d\xi_3,\\
	\d S_i=\d\ell_j\wedge\d\ell_k=H_jH_k\d\xi_j\d\xi_k,\quad(ijk)=(123)
\end{gather*}
In the $\bm e_k$ direction,
\begin{gather*}
	\int F_i\d S_i=\pv{F_iH_jH_k}{\ell_i}\d\ell_i\d\xi_j\d\xi_k,\\
	\Rightarrow\quad\oint_{\partial V}\ko5\bm F\cdot\d\bm S=\sum_{ijk}\pv{F_iH_jH_k}{\xi_i}\d\xi_i\d\xi_j\d\xi_k.
\end{gather*}
Then,
$$\div\bm F=\frac1{H_1H_2H_3}\sum_{ijk}\pp {\xi_i}F_iH_jH_k.$$
\paragraph{Curl} Follow the definition\index{curl}
$$(\curl\bm F)\cdot\uvec n:=\lim_{S\to0}\frac1S\oint_{\partial S}\ko5\bm F\cdot\d\bm\ell.$$
Take $\uvec n=\uvec e_i$, $S_i$ as square origin at $(\xi_j,\xi_k)$ with a delta $(\d\ell_j,\d\ell_k)$, thus
\begin{gather*}
	\d S_i=\d\ell_j\wedge\d\ell_k=H_jH_k\d\xi_j\d\xi_k.\\
	\oint_{\partial S_i}\ko5\bm F\cdot\d\bm\ell=\pv{F_kH_k}{\ell_j}\d\ell_j\d\xi_k-\pv{F_jH_j}{\ell_k}\d\ell_k\d\xi_j.
\end{gather*}
By using antisymmetric tensor $\varepsilon_{ijk}$
\begin{align*}
	\varepsilon_{ijk}=\begin{cases}
		1,  & (ijk)=(123), \\
		-1, & (ijk)=(321), \\
		0,  & \text{otherwise}.
	\end{cases}
\end{align*}
We can simplify the formula
\begin{align*}
	\curl\bm F & =\sum_{ijk}\varepsilon_{ijk}\uvec e_i\frac{\partial}{H_jH_k\partial\xi_j}F_kH_k \\
	                  & =\frac1{H_1H_2H_3}\begin{vmatrix}
		H_1\bm e_1             & H_2\bm e_2             & H_3\bm e_3             \\
		\partial/\partial\xi_1 & \partial/\partial\xi_2 & \partial/\partial\xi_3 \\
		H_1F_1                 & H_2F_2                 & H_3F_3
	\end{vmatrix}
\end{align*}
\paragraph{Laplacian} Follow the definition\index{Laplacian}
\begin{align*}
	\laplacian:= & \div\nabla=\div\sum_{i=1}^3\bm e_i\frac\partial{H_i\partial\xi_i} \\
	=          & \frac1{H_1H_2H_3}\sum_{ijk}\pp {\xi_i}H_jH_k\frac{\partial}{H_i\partial\xi_i}
\end{align*}

\subsubsection{Fourier Transformation\index{Fourier Transformation}}
We shall define the \textbf{inner product}\index{inner product} in $[-\pi,\pi]$ of real functions $f(x)$ and $g(x)$:
$$\brkt fg:=\int_{-\pi}^\pi f(x)g(x)\d x.$$
Omit math proofs\footnote{In math, something may not be strictly valid, but they're indeed useful \textbf{in physics}.}, we take it for granted the definition is \textbf{complete} in physics.\\
On the other hand:
\begin{gather*}
	\int_{-\pi}^\pi\sin nx\d x=\int_{-\pi}^\pi\cos nx\d x=0.\\
	\begin{aligned}
		\int_{-\pi}^\pi\sin nx\sin mx\d x & =\frac12\int_{-\pi}^\pi\cos(n-m)x-\cos(n+m)x\d x=\pi\delta_{nm}; \\
		\int_{-\pi}^\pi\sin nx\cos mx\d x & =\frac12\int_{-\pi}^\pi\sin(n+m)x+\sin(n-m)x\d x=0;              \\
		\int_{-\pi}^\pi\cos nx\cos mx\d x & =\frac12\int_{-\pi}^\pi\cos(n-m)x+\cos(n+m)x\d x=\pi\delta_{nm}.
	\end{aligned}
\end{gather*}
Therefore, The set
$$\{1,\sin nx,\cos nx\,|\,n\in\NN_+\}=\{1,\sin x,\cos x,\sin2x,\ldots\}$$
consists a set of bases in $[-\pi,\pi]$. Normalize it:
$$\Bigl\{\frac1{\sqrt{2\pi}},\frac{\sin x}{\sqrt\pi},\frac{\cos x}{\sqrt\pi},\frac{\sin2x}{\sqrt\pi},\ldots\Bigr\}$$
Then, for $f(x)$ with period $2\pi$ can be expand
$$f(x)=\frac{a_0}2+\sum_{n=1}^\infty(a_n\cos nx+b_n\sin nx),$$
which is the \textbf{Fourier Expansion}\index{Fourier Expansion}, where
\begin{align*}
	a_n & =\frac1\pi\brkt f{\cos nx}=\frac1\pi\int_{-\pi}^\pi f(x)\cos nx\d x, \\
	b_n & =\frac1\pi\brkt f{\sin nx}=\frac1\pi\int_{-\pi}^\pi f(x)\sin nx\d x.
\end{align*}
Use $\e^{\i x}=\cos x+\i\sin x$, we can write
$$f(x)=\sum_{n=-\infty}^{+\infty}c_n\e^{\i nx},\quad c_n=\frac1{2\pi}\int_{-\pi}^\pi f(x)\e^{-\i nx}\d x.$$
For any period $T=\lambda, k_0=2\pi/\lambda$ function $f(x)$
$$f(x)=\sum_{n=-\infty}^{+\infty}c_n\e^{\i nk_0x},\quad c_n=\frac1\lambda\int_{-\lambda/2}^{\lambda/2} f(x)\e^{-\i nk_0x}\d x.$$
When $f(x)$ is non-period, Fourier Expansion can't be taken into use.\\
However, we could take very LARGE $\lambda=2N,\Delta k=\frac{2\pi}\lambda,k:=n\Delta k$,
$$f(x)=\sum_k\biggl[\frac{\Delta k}{2\pi}\int_{-N}^Nf(x)\e^{-\i kx}\d x\biggr]\e^{\i kx}.$$
When $N\to\infty,\Delta k\to0$, define
$$\hat f(k):=\intR f(x)\e^{-\i kx}\d x,$$
then,
$$f(x)=\frac1{2\pi}\intR\hat f(k)\e^{\i kx}\d k,$$
which is the Fourier Transformation. In Shou's Note, for symmetrization,
\begin{gather*}
	\hat f(k)=\FT[f(x)]=\frac1{\sqrt{2\pi}}\intR f(x)\e^{-\i kx}\d x,\\
	f(x)=\FT\inv[\hat f(k)]=\frac1{\sqrt{2\pi}}\intR \hat f(k)\e^{\i kx}\d k.
\end{gather*}

\subsubsection{Legendre Function}
The solution of
$$(1-x^2)\dv[2] yx-2x\dv yx+\Bigb{J-\frac{m^2}{1-x^2}}y=0,$$
is $J=\ell(\ell+1)$ and $\ell=0,1,2,\ldots;m=0,\pm 1,\ldots,\pm\ell$,\footnote{In the calculation, we neglect $m$'s sign, i.e. in the formula, $m=\abs{m}$.}
$$y=AP_\ell^m(x),$$
where the Legendre Function $P_n^m(x)$\index{Legendre Function}
$$P_n^m(x)=P_n^{-m}(x)=(1-x^2)^{m/2}\dd[m]{x}P_n(x),$$
and the Legendre Polynomial $P_n(x)$\index{Legendre Polynomial}
$$P_n(x)=\frac1{2^nn!}\dd[n]{x}(x^2-1)^n.$$
Which is the solution of
$$(1-x^2)y''-2xy'+n(n+1)y=0.$$
The integral
\begin{gather*}
	\int_{-1}^1P_nP_{n'}\d x=\frac2{2n+1}\delta_{nn'},\\
	\int_{-1}^1P_n^mP_{n'}^{m'}\d x=\frac2{2n+1}\frac{(n+m)!}{(n-m)!}\delta_{nn'}\delta_{mm'}.
\end{gather*}
\subsubsection{Laguerre Function}
The solution of
$$\dv[2] u\xi=\Bigl[1-\frac N\xi+\frac{\ell(\ell+1)}{\xi^2}\Bigr]u,$$
is $N=2n$ and $n=1,2,\ldots;\ell=1,2,\ldots,n-1$,
$$R=A\rho^\ell \e^{-\rho/2}L_q^p(\rho),$$
where the Laguerre Function $L_q^p(x)$\index{Laguerre Function}
$$L_q^p(x)=(-1)^p\dd[p]{x}L_{p+q}(x),$$
and the Lagrange Polynomial $L_q(x)$\index{Lagrange Polynomial}
$$L_q(x)=\frac{\e^x}{q!}\dd[q]{x}\frac{x^q}{\e^x}.$$
Which is the solution of
$$xy''+(1-x)y'+qy=0.$$
The integral
\begin{gather*}
	\intRplus L_qL_{q'}\e^{-x}\d x=\delta_{qq'},\\
	\intRplus L_q^pL_{q'}^{p'}x^p\e^{-x}\d x=\frac{(p+q)!}{q!}\delta_{pp'}\delta_{qq'},\\
	\intRplus L_q^pL_{q'}^{p'}x^{p+1}\e^{-x}\d x=(2q+p+1)\frac{(p+q)!}{q!}\delta_{pp'}\delta_{qq'}.
\end{gather*}
