\chapter{基本假设}

\section{数学基础}

\begin{definition}
    {线性空间}{linear space}
    给定复数域$\CC$和集合$V$，若
    
    加法$+:V\times V\to V$满足：
    \begin{itemize}
        \item 结合律：$\forall\ket a,\ket b,\ket c\in V,$
        \begin{equation}
			(\ket a+\ket b)+\ket c=\ket a+(\ket b+\ket c),
		\end{equation}
        \item 零元$\ket 0\in V$：$\forall\ket a\in V,$
        \begin{equation}
			\ket a+\ket 0=\ket a,
		\end{equation}
        \item 逆元：$\forall\ket a\in V$，均存在$-\ket a\in V$
		\begin{equation}
			\ket a+(-\ket a)=\ket 0,
		\end{equation}
        \item 交换律：$\forall\ket a,\ket b\in V$
		\begin{equation}
			\ket a+\ket b=\ket b+\ket a;
		\end{equation}
    \end{itemize}
    数乘$\cdot:\CC\times V\to V$满足：
    \begin{itemize}
        \item 单位元$1\in\CC$：$\forall\ket a\in V,$
		\begin{equation}
			1\ket a=\ket a,
		\end{equation}
        \item 结合律：$\forall\lambda,\mu\in\CC,\,\ket a\in V,$
		\begin{equation}
			(\lambda\mu)\ket a=\lambda(\mu \ket a),
		\end{equation}
        \item 分配率1：$\forall\lambda,\mu\in\CC,\,\ket a\in V,$
		\begin{equation}
			(\lambda+\mu)\ket a=\lambda \ket a+\mu \ket a,
		\end{equation}
        \item 分配率2：$\forall\lambda\in\CC,\,\ket a,\ket b\in V,$
		\begin{equation}
			\lambda(\ket a+\ket b)=\lambda \ket a+\lambda \ket b.
		\end{equation}
    \end{itemize}
    则称$V$在$\CC$上构成一个线性空间(linear space)。
\end{definition}

\begin{definition}
	{内积}{inner product}
	给定线性空间，若存在映射$\brkt{\cdot}{\cdot}:V\times V\to\CC$，使得$\forall\ket a,\ket b,\ket c\in V,\,\lambda\in\CC$满足：
	\begin{itemize}
		\item 共轭对称性：
		\begin{equation}
			\brkt ba=\brkt ab\cj,
		\end{equation}
		\item 线性：
		\begin{subequations}
			\begin{gather}
				\bra a(\ket b+\ket c)=\brkt ab+\brkt ac, \\
				(\bra a+\bra b)\ket c=\brkt ac+\brkt bc, \\
				\bra a(\lambda\ket b)=\lambda\brkt ab.
			\end{gather}
		\end{subequations}
		\item 正定性：
		\begin{equation}
			\brkt aa\geqslant0,
		\end{equation}
		且$\brkt aa=0\iff\ket a=\ket 0.$
	\end{itemize}
	则称$\brkt{\cdot}{\cdot}$为内积(inner product)，$V$构成内积空间(inner product space)。
\end{definition}

\begin{definition}
	{对偶空间}{dual space}
	可以将内积视为映射
	\begin{equation}
		\bra a:V\to\CC,\quad\ket b\mapsto\brkt ab.
	\end{equation}
	连续线性泛函组成连续对偶空间$V^*$。有限维内积空间或完备的Hilbert空间中，Riesz表示定理保证每个连续线性泛函都可唯一地写成上述$\bra a$的形式。映射$\ket a\mapsto\bra a$是共轭线性的。
\end{definition}

\begin{definition}
	{Dirac符号}{Dirac notation}
	给定线性空间$V$，$\ket a\in V$称为右矢(ket vector)，而$\bra a\in V^*$称为左矢(bra vector)。
\end{definition}

\begin{definition}
	{范数}{norm}
	给定内积空间，定义范数为映射$\norm{\cdot}:V\to\RR$，使得$\forall\ket a,\ket b\in V,\,\lambda\in\CC$满足：
	\begin{itemize}
		\item 正定性：
		\begin{equation}
			\norm{\ket a}\geqslant0,
		\end{equation}
		且$\norm{\ket a}=0\iff\ket a=\ket 0;$
		\item 齐次性：
		\begin{equation}
			\norm{\lambda\ket a}=\abs\lambda\norm{\ket a};
		\end{equation}
		\item 三角不等式：
		\begin{equation}
			\norm{\ket a+\ket b}\leqslant\norm{\ket a}+\norm{\ket b}.
		\end{equation}
	\end{itemize}
	则称$\norm{\cdot}$为范数(norm)，$V$构成赋范空间(normed space)。
\end{definition}

\begin{corollary}
	根据内积可以定义范数：
	\begin{equation}
		\norm{\ket a}:=\sqrt{\brkt a a}.
	\end{equation}
	称为内积诱导的范数。
	因此内积空间必然是赋范空间。
\end{corollary}

\begin{definition}
	{赋范空间的完备性}{completeness of normed space}
	赋范空间$V$若满足：对于任意Cauchy列$\{\ket{a_n}\}$，均存在$\ket a\in V$使得
	\begin{equation}
		\lim_{n\to\infty}\norm{\ket{a_n}-\ket a}=0,\quad
		(\text{记作$\lim_{n\to\infty}\ket{a_n}=\ket a$})
	\end{equation}
	则称$V$是完备的(complete)。
	完备的赋范空间称为Banach空间，
	完备的内积空间称为Hilbert空间。
\end{definition}

在本节的纯态描述中，一个物理态(physical state)由复Hilbert空间中的态矢量(state vector) $\ket\psi$表示。
且任意非零复数倍的态矢量表示同一物理态，为了方便，通常取归一化的态矢量，即$\brkt\psi\psi=1$。

\begin{postulate}
	{量子力学基本假设}{Basic Postulate of Quantum Mechanics}
	\begin{itemize}
		\item 物理系统的每一个纯态对应Hilbert空间中的一条射线，即非零态矢量$\ket\psi$及其所有非零复数倍；混合态需用密度算符描述；
		\item 可观测量对应自伴算符（有限维时即Hermite矩阵），测量值由其谱给出。对纯离散谱可选正交归一本征基；连续谱需用广义本征态和谱积分处理；
		\item 对离散谱的理想投影测量，测得$a$的概率为$p_a=\bra\psi\hat P_a\ket\psi$，其中$\hat P_a$投影到对应本征子空间。若$p_a>0$，测后态为$\hat P_a\ket\psi/\sqrt{p_a}$；非简并时化为$\abs{\brkt{\psi_a}\psi}^2$；
		\item 态矢量随时间演化满足Schrödinger方程
		\[
			\i\hbar\p_t\ket\psi=\hat H\ket\psi,
		\]
		其中$\hat H$为系统的Hamilton算符。
	\end{itemize}
\end{postulate}




\begin{definition}{波函数}{Wave Function}
	波函数$\psi(\r)$刻画粒子在空间的概率分布．约定内积记号
	\[
		(\psi,\varphi)\equiv\int\psi^*(\r)\varphi(\r)\d\r,
	\]	
	波函数应是归一化的，即$(\psi,\psi)=1.$
\end{definition}
由Fourier展开，粒子的动量分布
\begin{align}
	\psi(\r)       & =\frac1{(2\pi\hbar)^{3/2}}\int\varphi(\bm p)\e^{\i\bm p\cdot\r/\hbar}\d\bm p; \\
	\varphi(\bm p) & =\frac1{(2\pi\hbar)^{3/2}}\int\psi(\r)\e^{-\i\bm p\cdot\r/\hbar}\d\bm r.
\end{align}
\begin{theorem}{不确定度关系}{Uncertainty Relation}
	% Heisenberg提出\textbf{不确定度关系}：
	对同一归一化量子态，位置$x$与相应正则动量$p_x$的标准差满足
	\begin{align}
		\delta x\,\delta p_x\geqslant\hbar/2.
	\end{align}
	证明见第 \pageref{Proof of UR} 页。
\end{theorem}
这里的不确定度指多次同样制备下测量结果的统计散布，不直接等同于测量仪器的误差。坐标表象中，由空间平移的生成元得到正则动量算符
\[
	\qo p=-\i\hbar\nabla,
\]
\begin{definition}{平均值}{Mean Value}
	力学量的平均值，比如势能$V(\r)$
	\[
		\avg V\equiv(\psi,V\psi)=\int\abs{\psi(\r)}^2V(\r)\d\bm r,
	\]
	动量的平均值
	\[
		\avg{\bm p}=(\psi,\qo p\psi)=\int\psi^*(\r)\qo p\psi(\r)\d\bm r.
	\]
	推导见第 \pageref{Derivation of Mean Value} 页。
\end{definition}
\begin{theorem}{Schrödinger方程}{Schrödinger Equation}
	波函数随时间的演化满足
	\begin{align}
		\i\hbar\pv{\psi}t=\hat H\psi.
	\end{align}
	对无自旋、非相对论粒子，在实标量势$V$中且无电磁矢势时，Hamilton算符
	\[
		\hat H=\frac{\hat p^2}{2m}+V.
	\] % =-\frac{\hbar^2}{2m}\nabla^2+V.\]
\end{theorem}
\begin{definition}{概率密度和概率流密度}{Probility Density}
	概率密度
	\[
		\rho(\r,t):=\psi^*(\r,t)\psi(\r,t),
	\]
	概率流密度
	\[
		\bm j(\r,t):=-\frac{\i\hbar}{2m}(\psi^*\nabla\psi-\psi\nabla\psi^*).
	\]
	由Schrödinger方程，二者有关系
	\[
		\pv\rho{t}=\psi^*\pv\psi{t}+\psi\pv{\psi^*}t=-\nabla\cdot\bm j.
	\]
\end{definition}
当$\hat H$不显含时间时，考虑其本征值$E_n$和本征函数$\psi_n(\r)$
\begin{align}
	\hat H\psi_n(\r)=E_n\psi_n(\r).
\end{align}
称为定态Schrödinger方程，代表能量有确定值，而在纯离散谱情况下，含时方程的通解可以写成定态的线性组合（连续谱部分改用积分）
\[
	\psi(\r,t)=\sum c_n\psi_n(\r)\e^{-\i E_nt/\hbar}.
\]
