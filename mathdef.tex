%% Delimiters: ordinary size by default, explicit sizes for taller material.
\makeatletter
\newcommand*{\mathdef@new}[1]{\expandafter\newcommand\csname#1\endcsname}
% #1 is the name prefix; #2, #3, #4 size opening, inner and closing delimiters.
% An inner Dirac bar or a division slash has no relation spacing.
\newcommand*{\mathdef@delimiters}[4]{%
    \mathdef@new{#1abs}[1]{#2\lvert##1#4\rvert}%
    \mathdef@new{#1norm}[1]{#2\lVert##1#4\rVert}%
    \mathdef@new{#1eval}[1]{{##1}#4\rvert}%
    \mathdef@new{#1floor}[1]{#2\lfloor##1#4\rfloor}%
    \mathdef@new{#1ceil}[1]{#2\lceil##1#4\rceil}%
    \mathdef@new{#1ave}[1]{#2\langle##1#4\rangle}%
    \mathdef@new{#1set}[2]{#2\{##1\,#3\vert\,##2#4\}}%
    \mathdef@new{#1division}[2]{{##1}#3/{##2}}%
    \mathdef@new{#1inp}[2]{#2\langle##1,##2#4\rangle}%
    \mathdef@new{#1bra}[1]{#2\langle##1#4\rvert}%
    \mathdef@new{#1ket}[1]{#2\lvert##1#4\rangle}%
    \mathdef@new{#1brkt}[2]{#2\langle##1#3\vert##2#4\rangle}%
    \mathdef@new{#1mel}[3]{#2\langle##1#3\vert##2#3\vert##3#4\rangle}%
    \mathdef@new{#1ktbr}[2]{#2\lvert##1#4\rangle#2\langle##2#4\rvert}%
}
% Plain ( ), [ ] and \{ \} already have ordinary-size spellings.
\newcommand*{\mathdef@sized}[4]{%
    \mathdef@delimiters{#1}{#2}{#3}{#4}%
    \mathdef@new{#1p}[1]{#2(##1#4)}%
    \mathdef@new{#1b}[1]{#2[##1#4]}%
    \mathdef@new{#1B}[1]{#2\{##1#4\}}%
}
\mathdef@delimiters{}{}{}{}
\mathdef@sized{big} \bigl \big \bigr
\mathdef@sized{Big} \Bigl \Big \Bigr
\mathdef@sized{bigg}\biggl\bigg\biggr
\mathdef@sized{Bigg}\Biggl\Bigg\Biggr
\makeatother
